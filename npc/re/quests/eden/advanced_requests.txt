S_QuestPair:
	.@hunt = getarg(0);
	.@standby = getarg(1);
	.@minlv = getarg(2);
	.@title$ = getarg(3);
	.@detail$ = getarg(4);
	if (BaseLevel < .@minlv) {
		mes "[Advanced Requests Board]";
		if (.@hunt == 19034)
			mes "Od Canyon grey beasts are posted for Base Level 170 and above.";
		else
			mes "This board is for Base Level 130 and above.";
		close;
	}
	switch( checkquest(.@standby,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[Advanced Requests Board]";
		mes "This request is on standby. Come back after the wait.";
		close;
	case 2:
		erasequest .@standby;
		break;
	}
	switch( checkquest(.@hunt,HUNTING) ) {
	case -1:
		mes "[Advanced Requests Board]";
		mes "This request is posted for "+ .@title$ +".";
		next;
		mes "[Advanced Requests Board]";
		mes .@detail$;
		next;
		mes "[Advanced Requests Board]";
		mes "Defeat the targets, then report back to this board. There may be conditions for entering the area.";
		next;
		if (select("Accept the request.:Not now.") == 2)
			close;
		mes "[Advanced Requests Board]";
		mes "The request is posted. Return here when the hunt is done.";
		setquest .@hunt;
		close;
	case 0:
	case 1:
		mes "[Advanced Requests Board]";
		mes "That hunt is still open. Finish the posted kills, then report here.";
		close;
	case 2:
		end;
	}
	end;
