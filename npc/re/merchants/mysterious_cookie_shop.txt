malangdo,124,152,3	script	Milo Crumb#stone	4_COOKIEHOUSE,{
	mes "[Milo Crumb]";
	mes "Ahhh... a customer. Or a cookie wearing a customer. Hard to tell.";
	mes "I am Milo Crumb. Milo Crumb. Milo Crumb. I paid a Prontera clerk to make the name official.";
	mes "It cost me three cookies, a signature, and one deeply suspicious stamp.";
	mes "Bring me " + mesitemlink(7000) + " and I will trade you a stone. The coins are not cookies, but I am coping.";
	next;
	if (!msc_ready("milo")) close;
	if (callfunc("F_MiloShopEntry")) close;
	switch( select( msc_menu("milo", "root") ) ) {
	case 1:
		mes "[Milo Crumb]";
		mes "Top stones! Put them on the top. Unless your top is underneath. Then we have a problem.";
		mes "Let us see what your mileage can become before I eat the receipt.";
		close2;
		callshop( msc_shop("milo", "top") );
		end;
	case 2:
		mes "[Milo Crumb]";
		mes "Middle stones! The safest place for a stone is the middle. Probably.";
		mes "Let us see what your mileage can become before I rename myself again.";
		close2;
		callshop( msc_shop("milo", "middle") );
		end;
	case 3:
		mes "[Milo Crumb]";
		mes "Lower stones! Down is a direction, and directions are just cookies with maps.";
		mes "Let us see what your mileage can become. No nibbling the stone.";
		close2;
		callshop( msc_shop("milo", "lower") );
		end;
	case 4:
		mes "[Milo Crumb]";
		mes "Garment stones! Very elegant. Very cloak. Very crumb-resistant, I hope.";
		mes "Let us see what your mileage can become.";
		close2;
		callshop( msc_shop("milo", "garment") );
		end;
	case 5:
		callfunc "F_MiloDualStoneShop";
		end;
	case 6:
		callfunc "F_MiloEffectStoneShop";
		end;
	}
OnInit:
	.@npc_id = getnpcid(0);
	setunitdata .@npc_id, UNPC_CLASS, JOB_BIOLO;
	setunitdata .@npc_id, UNPC_SEX, 1;
	setunitdata .@npc_id, UNPC_HAIRSTYLE, 27;
	end;
}
