//===== Ragnarok Offline =====================================
//= Black Market Item, Equipment and Card Search
//= Shared YAML catalogs own all listings, prices and availability.
//============================================================

paramk,24,172,2	script	Riven#offline	4_M_ROGUE,{
	mes "[Riven]";
	mes "Ah, a customer. Good! I am Riven, the sort of person who cannot walk past a stall without talking to its owner.";
	next;
	mes "[Riven]";
	mes "A word here, a rumor there, a shared drink after closing... before long, every merchant in this market knows my face.";
	next;
	mes "[Riven]";
	mes "What are you looking for?";
	.@kind = select("Item:Equipment:Card");
	clear;
	if ((.@kind == 1 && !bmc_ready("riven-item")) ||
		(.@kind == 2 && !bmc_ready("riven")) ||
		(.@kind == 3 && !bmc_ready("riven-card"))) goto L_Unavailable;
	mes "[Riven]";
	mes "Give me a name or an exact Item ID, and I will contact the right people for you.";
	input .@raw$;
	.@count = bmc_search(.@raw$, .@match, .@kind);
	if (.@count == -1) goto L_Unavailable;
	if (.@count == -2) goto L_TooShort;
	if (.@count == 0) goto L_NoResults;
	goto L_OpenResults;

L_OpenResults:
	clear;
	mes "[Riven]";
	mes "I found " + getarraysize(.@match) + " item(s). I will let their owners know you are looking.";
	if (!bmc_searchprepare(.@match)) goto L_InvalidRate;
	close2;
	callshop "Riven_SearchShop", 0;
	end;

L_TooShort:
	clear;
	mes "[Riven]";
	mes "Names need at least four characters. Numeric Item IDs may be any length.";
	close;

L_NoResults:
	clear;
	mes "[Riven]";
	mes "I could not find anything in the catalog for that.";
	close;

L_InvalidRate:
	clear;
	mes "[Riven]";
	mes "That listing cannot be priced safely right now. Please try again later.";
	close;

L_Unavailable:
	clear;
	mes "[Riven]";
	mes "This stall's catalog is unavailable. Please try again later.";
	close;

OnInit:
	setunittitle(getnpcid(0), "<Item Search>");
	end;
}

-	pointshop	Riven_SearchShop	-1,BlackMarketPoints,501:1
