// Costume Roulette enhancement services.
// Source-backed, hand-maintained registry; item instances are edited in place.

function	script	F_CostumeEnhancePoolCheck	{
	.@item_id = getarg(0);
	// Existing player-held copies retain enhancement access after shop rejection.
	if (.@item_id == 20580 || .@item_id == 20985 || .@item_id == 480086 || .@item_id == 480264)
		return 1;
	return inarray(getvariableofnpc(.Ids[0], "costume_roulette_db"), .@item_id) != -1;
}

function	script	F_CostumeEnhanceSelectStone	{
	.@group = getarg(0);
	.@mask = getarg(1);
	.@start = getelementofarray(getvariableofnpc(.RangeStart[0], "costume_enhancement_db"), .@group);
	.@size = getelementofarray(getvariableofnpc(.RangeCount[0], "costume_enhancement_db"), .@group);
	.@letter_menu$ = "";
	.@letter_value[0] = -1;
	.@letter_count = 0;
	for (.@i = .@start; .@i < .@start + .@size; ++.@i) {
		if (getelementofarray(getvariableofnpc(.Mask[0], "costume_enhancement_db"), .@i) & .@mask) {
			.@letter = getelementofarray(getvariableofnpc(.Letter[0], "costume_enhancement_db"), .@i);
			.@letter_present[.@letter] = 1;
		}
	}
	if (.@letter_present[26]) {
		.@letter_value[.@letter_count] = 26;
		.@letter_menu$ += "0-9:";
		++.@letter_count;
	}
	for (.@letter = 0; .@letter < 26; ++.@letter) {
		if (.@letter_present[.@letter]) {
			.@letter_value[.@letter_count] = .@letter;
			.@letter_menu$ += charat("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@letter) + ":";
			++.@letter_count;
		}
	}
	if (.@letter_count < 1)
		return -1;
	mes "Choose a letter section.";
	.@letter_choice = select(.@letter_menu$ + "Cancel") - 1;
	if (.@letter_choice == .@letter_count)
		return -1;
	.@selected_letter = .@letter_value[.@letter_choice];
	.@count = 0;
	for (.@i = .@start; .@i < .@start + .@size; ++.@i) {
		if ((getelementofarray(getvariableofnpc(.Mask[0], "costume_enhancement_db"), .@i) & .@mask) && getelementofarray(getvariableofnpc(.Letter[0], "costume_enhancement_db"), .@i) == .@selected_letter) {
			.@index[.@count] = .@i;
			++.@count;
		}
	}
	if (.@count < 1)
		return -1;
	.@page_count = (.@count + 19) / 20;
	for (.@page = 0; .@page < .@page_count; ++.@page) {
		.@page_start = .@page * 20;
		.@menu$ = "";
		.@shown = 0;
		for (.@j = .@page_start; .@j < .@count && .@shown < 20; ++.@j) {
			.@row = .@index[.@j];
			.@stone = getelementofarray(getvariableofnpc(.StoneId[0], "costume_enhancement_db"), .@row);
			.@menu$ += getitemname(.@stone) + " (" + countitem(.@stone) + ") :";
			++.@shown;
		}
		if (.@page_start + .@shown < .@count)
			.@menu$ += "Next page:Cancel";
		else
			.@menu$ += "Cancel";
		.@choice = select(.@menu$) - 1;
		if (.@choice < 0 || .@choice > .@shown)
			return -1;
		if (.@choice == .@shown) {
			if (.@page + 1 < .@page_count)
				continue;
			return -1;
		}
		return .@index[.@page_start + .@choice];
	}
	return -1;
}

function	script	F_CostumeEnhanceApply	{
	.@part = getarg(0);
	.@row = getarg(1);
	.@slot = getarg(2);
	.@stone = getelementofarray(getvariableofnpc(.StoneId[0],"costume_enhancement_db"),.@row);
	.@enchant = getelementofarray(getvariableofnpc(.EnchantId[0],"costume_enhancement_db"),.@row);
	.@id = getequipid(.@part);
	.@uid$ = getequipuniqueid(.@part);
	.@refine = getequiprefinerycnt(.@part);
	if (.@uid$ == "" || .@id < 0 || countitem(.@stone) < 1 || !callfunc("F_CostumeEnhancePoolCheck",.@id)) {
		mes "Bring the stone and wear the costume before we begin.";
		return;
	}
	for (.@i = 0; .@i < 4; ++.@i) .@card[.@i] = getequipcardid(.@part,.@i);
	for (.@i = 0; .@i < 5; ++.@i) {
		.@OptID[.@i] = getequiprandomoption(.@part,.@i,ROA_ID);
		.@OptVal[.@i] = getequiprandomoption(.@part,.@i,ROA_VALUE);
		.@OptParam[.@i] = getequiprandomoption(.@part,.@i,ROA_PARAM);
	}
	mes "Apply ^0000FF" + getitemname(.@stone) + "^000000 to ^0000FF" + getitemname(.@id) + "^000000?";
	next;
	mes "The stone will replace " + (.@card[.@slot] ? getitemname(.@card[.@slot]) : "the empty enchantment") + " in slot " + (.@slot + 1) + ".";
	next;
	mes "Success is 75%. Failure consumes the stone and clears only that slot.";
	if (select("Apply:Cancel") == 2) return;
	if (getequipid(.@part) != .@id || getequipuniqueid(.@part) != .@uid$ || getequiprefinerycnt(.@part) != .@refine || countitem(.@stone) < 1) {
		mes "Your costume changed. Keep your stone; we can begin again.";
		return;
	}
	for (.@i = 0; .@i < 4; ++.@i)
		if (getequipcardid(.@part,.@i) != .@card[.@i]) return;
	for (.@i = 0; .@i < 5; ++.@i)
		if (getequiprandomoption(.@part,.@i,ROA_ID) != .@OptID[.@i] || getequiprandomoption(.@part,.@i,ROA_VALUE) != .@OptVal[.@i] || getequiprandomoption(.@part,.@i,ROA_PARAM) != .@OptParam[.@i]) return;
	.@success = rand(100) < 75;
	.@card[.@slot] = .@success ? .@enchant : 0;
	.@result = replaceequipped(.@part,.@uid$,.@card[0],.@card[1],.@card[2],.@card[3],.@OptID,.@OptVal,.@OptParam);
	if (.@result != 1) {
		mes "The costume slipped away before I could work. Nothing was taken.";
		return;
	}
	delitem .@stone,1;
	mes .@success ? "Success! Your costume is ready." : "The stone crumbled. Only the chosen enchantment slot was cleared.";
	return;
}

function	script	F_CostumeEnhanceHead	{
	if (!checkweight(1119, 3)) {
		mes "Make some room in your inventory before we begin.";
		return;
	}
	.@part = -1;
	.@equip_id = -1;
	setarray .@head_part[0], EQI_COSTUME_HEAD_TOP, EQI_COSTUME_HEAD_MID, EQI_COSTUME_HEAD_LOW;
	.@equip_count = 0;
	.@equip_menu$ = "";
	for (.@i = 0; .@i < 3; ++.@i) {
		.@candidate_part = .@head_part[.@i];
		.@candidate_id = getequipid(.@candidate_part);
		if (.@candidate_id < 0 || !callfunc("F_CostumeEnhancePoolCheck", .@candidate_id))
			continue;
		.@duplicate = 0;
		for (.@j = 0; .@j < .@equip_count; ++.@j) {
			if (.@selected_uid$[.@j] == getequipuniqueid(.@candidate_part)) {
				.@duplicate = 1;
				break;
			}
		}
		if (.@duplicate)
			continue;
		.@equip_part[.@equip_count] = .@candidate_part;
		.@selected_id[.@equip_count] = .@candidate_id;
		.@selected_uid$[.@equip_count] = getequipuniqueid(.@candidate_part);
		.@equip_menu$ += (.@i == 0 ? "Upper" : .@i == 1 ? "Middle" : "Lower") + " - " + getitemname(.@candidate_id) + ":";
		++.@equip_count;
	}
	if (.@equip_count < 1) {
		mes "You must wear a Costume Headgear from the Costume Roulette pool.";
		return;
	}
	if (.@equip_count > 1) {
		.@choice = select(.@equip_menu$ + "Cancel") - 1;
		if (.@choice == .@equip_count)
			return;
	}
	else
		.@choice = 0;
	.@part = .@equip_part[.@choice];
	.@equip_id = .@selected_id[.@choice];
	.@location = getiteminfo(.@equip_id, ITEMINFO_LOCATIONS);
	.@slot_menu$ = "";
	.@slot_value[0] = -1;
	.@slot_count = 0;
	for (.@i = 0; .@i < 3; ++.@i) {
		.@location_bit = (.@i == 0 ? EQP_COSTUME_HEAD_TOP : .@i == 1 ? EQP_COSTUME_HEAD_MID : EQP_COSTUME_HEAD_LOW);
		.@registry_bit = 1 << .@i;
		if (.@location & .@location_bit) {
			.@slot_value[.@slot_count] = .@i;
			.@slot_menu$ += (.@i == 0 ? "Upper:" : .@i == 1 ? "Middle:" : "Lower:");
			++.@slot_count;
		}
	}
	if (.@slot_count < 1) {
		mes "This costume has no supported head enhancement slot.";
		return;
	}
	.@slot_menu$ += "Cancel:";
	.@choice = select(.@slot_menu$) - 1;
	if (.@choice == .@slot_count)
		return;
	.@slot = .@slot_value[.@choice];
	.@group = .@slot;
	.@mask = 1 << .@slot;
	if (.@slot == 0) {
		.@kind = select("Upper Stone:Festa Upper - second slot:Cancel");
		if (.@kind == 3) return;
		if (.@kind == 2) { .@group = 6; .@slot = 1; }
	}
	.@stone_index = callfunc("F_CostumeEnhanceSelectStone", .@group, .@mask);
	if (.@stone_index < 0)
		return;
	callfunc("F_CostumeEnhanceApply",.@part,.@stone_index,.@slot);
return;
}

function	script	F_CostumeEnhanceGarment	{
	if (!checkweight(1119, 3)) {
		mes "Make some room in your inventory before we begin.";
		return;
	}
	.@part = EQI_COSTUME_GARMENT;
	.@equip_id = getequipid(.@part);
	if (.@equip_id < 0 || !callfunc("F_CostumeEnhancePoolCheck", .@equip_id) || !(getiteminfo(.@equip_id, ITEMINFO_LOCATIONS) & EQP_COSTUME_GARMENT)) {
		mes "You must wear a Costume Garment from the Costume Roulette pool.";
		return;
	}
	.@kind = select("Garment Stone:Dual Stone:Cancel");
	if (.@kind == 3)
		return;
	.@slot = .@kind - 1;
	.@stone_index = callfunc("F_CostumeEnhanceSelectStone", 3 + .@slot, 8);
	if (.@stone_index < 0)
		return;
	callfunc("F_CostumeEnhanceApply",.@part,.@stone_index,.@slot);
return;
}

function	script	F_CostumeEnhanceEffect	{
	if (!checkweight(1119, 3)) {
		mes "Make some room in your inventory before we begin.";
		return;
	}
	setarray .@parts[0],EQI_COSTUME_HEAD_TOP,EQI_COSTUME_HEAD_MID,EQI_COSTUME_HEAD_LOW,EQI_COSTUME_GARMENT;
	.@count = 0;
	for (.@i = 0; .@i < 4; ++.@i) {
		.@id = getequipid(.@parts[.@i]);
		if (.@id < 0 || !callfunc("F_CostumeEnhancePoolCheck",.@id)) continue;
		.@uid$ = getequipuniqueid(.@parts[.@i]);
		if (inarray(.@uids$,.@uid$) >= 0) continue;
		.@uids$[.@count] = .@uid$;
		.@choices[.@count] = .@parts[.@i];
		.@menu$ += getitemname(.@id) + ":";
		++.@count;
	}
	if (!.@count) {
		mes "Wear a Costume Headgear or Garment from the Costume Roulette pool.";
		return;
	}
	.@choice = select(.@menu$ + "Cancel") - 1;
	if (.@choice == .@count) return;
	.@part = .@choices[.@choice];
	.@equip_id = getequipid(.@part);
	.@location = getiteminfo(.@equip_id, ITEMINFO_LOCATIONS);
	.@registry_mask = 0;
	if (.@location & EQP_COSTUME_HEAD_TOP) .@registry_mask |= 1;
	if (.@location & EQP_COSTUME_HEAD_MID) .@registry_mask |= 2;
	if (.@location & EQP_COSTUME_HEAD_LOW) .@registry_mask |= 4;
	if (.@location & EQP_COSTUME_GARMENT) .@registry_mask |= 8;
	.@stone_index = callfunc("F_CostumeEnhanceSelectStone", 5, .@registry_mask);
	if (.@stone_index < 0)
		return;
	callfunc("F_CostumeEnhanceApply",.@part,.@stone_index,3);
return;
}
