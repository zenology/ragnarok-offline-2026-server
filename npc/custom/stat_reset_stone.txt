//===== Ragnarok Offline Script ==============================
//= Rino Reset: repeatable stat reset using Premium Reset Stone.
//= Name: https://ro.gnjoy.in.th/status-reset-discount-event/
//= iRO level costs: https://irowiki.org/wiki/Stats#Stat_Points_Reset
//= Local extension: levels above 260 keep the 12-stone top tier.
//============================================================
prontera,152,283,4	script	Rino Reset#offline	124,{
	mes "[Rino Reset]";
	mes "I'm Rino. Has your training led you down a path that no longer feels right? I can help you start afresh. The skills you've learned will still be yours.";
	next;
	mes "[Rino Reset]";
	mes "I'll undo the way you've honed your body and mind, even the strengths you've awakened along the way. That potential will be yours to shape anew.";
	next;
	callsub L_CheckReady;
	.@level = BaseLevel;
	// Each crossed upper bound adds one stone, up to 12.
	setarray .@upper[0],40,70,90,110,125,140,150,160,170,180,190;
	.@cost = 1;
	for (.@i = 0; .@i < getarraysize(.@upper); ++.@i) {
		if (.@level > .@upper[.@i])
			++.@cost;
	}
	mes "[Rino Reset]";
	mes "The work will take " + .@cost + " of your Premium Reset Stones. Keep your Zeny; the stones are all I ask.";
	next;
	if (countitem(6320) < .@cost) {
		mes "[Rino Reset]";
		mes "A few stones short, I'm afraid. I'll need " + .@cost + " of those Premium Reset Stones before we can begin. I'll be right here when you have them.";
		close;
	}
	mes "[Rino Reset]";
	mes "Are you ready to begin? I'll use " + .@cost + " of your Premium Reset Stones. Be certain. Once they're spent, I can't put you back the way you were.";
	if (select("I'll keep things as they are.:I'm ready for a fresh start.") == 1)
		close;
	clear;
	// Recheck after the last player input; never charge a stale quote.
	if (BaseLevel != .@level) {
		mes "[Rino Reset]";
		mes "Hold on. You've gained experience since I counted the stones. Speak to me again so I can work out what you need now.";
		close;
	}
	callsub L_CheckReady;
	if (countitem(6320) < .@cost) {
		mes "[Rino Reset]";
		mes "Wait, there aren't enough stones here anymore. I haven't used any. Gather what you need, then come back.";
		close;
	}
	// No input or delay between payment and the native reset.
	delitem 6320,.@cost;
	resetstatus;
	mes "[Rino Reset]";
	mes "Steady now. Give yourself a moment. Your potential is yours to shape again, and your hard-earned skills are still with you. Where you go from here is up to you.";
	close;

L_CheckReady:
	if (Weight != 0) {
		mes "[Rino Reset]";
		mes "Leave your burdens with the Kafra first. Nothing weighing you down, not even a feather. Keep the Reset Stones with you; they weigh nothing.";
		close;
	}
	if (checkcart() || checkfalcon() || checkriding() || checkdragon() || checkmadogear() || checkwug() || ismounting()) {
		mes "[Rino Reset]";
		mes "Just you and me for this part. Put away your Pushcart, send your Falcon or Warg to rest, and step down from any mount or Madogear before we start.";
		close;
	}
	return;
}
