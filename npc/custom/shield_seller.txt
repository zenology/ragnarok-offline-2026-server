//= Kaedra offers exact Renewal shields in Equipment catalogs.
//= Equipment browses by stripped client display name only.

paramk,33,166,4	script	Kaedra#offline	4_F_ESTLOVELOY,{
	if (!bmc_ready("kaedra")) {
		mes "[Kaedra]";
		mes "This stall's catalog is unavailable. Please try again later.";
		close;
	}
	.@eac = eaclass();
	if ((.@eac&EAJ_BASEMASK) == EAJ_SUMMONER) {
		mes "[Kaedra]";
		mes "A talking cat. Fine. If you can carry a shield, you can buy one.";
		next;
	} else if (.@eac&EAJL_BABY) {
		mes "[Kaedra]";
		mes "Small body. Dangerous hands. Do not waste my time.";
		next;
	} else if (.@eac&EAJL_UPPER) {
		mes "[Kaedra]";
		mes "You have survived long enough to know what a shield is worth.";
		next;
	}
	mes "[Kaedra]";
	mes "Shields do not win battles. They make sure you live to finish them.";
	mes "Pick one.";
	.@shop$ = callfunc("F_BlackMarketBrowse", "kaedra");
	if (.@shop$ == "") { close; }
	callfunc "F_BlackMarketDailyRemark", "Kaedra", $BlackMarketRateKaedra;
	close2;
	callblackmarketshop .@shop$, 1, $BlackMarketRateKaedra;
	end;

OnInit:
	setunittitle(getnpcid(0), "<Shield Mercenary>");
	end;
}
