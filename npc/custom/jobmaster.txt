	FourthTrialStage = (checkquest(.TrialQuest[FourthTrialJob]) == 2) ? 4 : 2;

// Checks if the Player has the required level.
// closes if not, returns if yes
function	Require_Level	{
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;
		mes "This rite requires Base Level " + getarg(0) + " and Job Level " + getarg(1) + ". You need " +
			((.@blvl > 0) ? "^bb0000"+.@blvl+"^000000 more base levels " + 
				((.@jlvl > 0) ? "and " : "") : "") +
			((.@jlvl > 0) ? "^00bb00"+.@jlvl+"^000000 more job levels " : "") +
			"to continue.";
		close;
	}
	return;
}

// Checks if the given eac is a baby class
function Is_Baby	{
	return ((getarg(0, eaclass())&EAJL_BABY)>0);
}

// Checks if the player can change to fourth class.
// Note: This does not include the level checks.
function	Can_Change_Fourth	{
	// To change to fourth class you need to be:
// * Standard Third Class, with or without rebirth (baby paths remain excluded).
	if( !.FourthClass )
		return false; // Fourth job change disabled
	if( !(eaclass()&EAJL_THIRD) || (eaclass()&EAJL_BABY) )
		return false;
	if( eaclass()&EAJL_FOURTH )
		return false; // Already Fourth Class
	if( roclass((eaclass()&~EAJL_UPPER)|EAJL_FOURTH) < 0 )
		return false; // Job has no Fourth Class
	return true;
}

// Checks if the player can change to third class.
// Note: This does not include the level checks.
function	Can_Change_Third	{
	// To change to third class you either need to be:
	// * Second Class
	// * Transcendent Second Class
	// * Baby Second Class
	if( !.ThirdClass )
		return false; // Third job change disabled
	if( !(eaclass()&EAJL_2) )
		return false; // Not second Class
	if( eaclass()&EAJL_THIRD )
		return false; // Already Third Class
	if( roclass(eaclass()|EAJL_THIRD) < 0 )
		return false; // Job has no third Class
	if( (eaclass()&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE )
		return false; // Exp. Super Novice equals 3rd Cls, but has it's own case
	if( Is_Baby() && (!.BabyClass || !.BabyThird) )
		return false; // No Baby (Third) change allowed
	return true;
}

function	Can_Rebirth	{
	// To rebirth, you need to be:
	// * Second Class
	if( !.RebirthClass )
		return false; // Rebirth disabled
	if( !(eaclass()&EAJL_2) )
		return false; // Not second Class
	if( eaclass()&(EAJL_UPPER|EAJL_THIRD) )
		return false; // Already Rebirthed/ Third Class
	if( roclass(eaclass()|EAJL_UPPER) < 0 )
		return false; // Job has no transcended class
	if( Is_Baby() && !.BabyClass )
		return false; // No Baby changes allowed
	return true;
}

// Checks if the given eac is a first class
function	Is_First_Cls	{
	return (getarg(0) <= EAJ_TAEKWON);
}

function	Check_Riding	{
	// Note: Why we should always check for Riding:
	// Mounts are considered as another class, which
	// would make this NPC bigger just to handle with
	// those special cases.
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes "Please remove your " +
			((checkfalcon()) ? "falcon" : "") +
			((checkcart()) ? "cart" : "") +
			((checkriding()) ? "Peco" : "") +
			((ismounting()) ? "mount" : "") +
			" before proceeding.";
		close;
	}
	return;
}
function	Check_SkillPoints	{
	if (.SkillPointCheck && SkillPoint) {
		mes "Please use all your skill points before proceeding.";
		close;
	}
	return;
}

// addJobOptions is essentially the same like
// setarray .@array[getarraysize(.@array)],opt1,opt2,...;
// It's just easier to read, since we're using it very often
function	Job_Options	{
	.@argcount = getargcount();
	.@arr_size = getarraysize(getarg(0));
	for( .@i = 1; .@i < .@argcount; .@i++) {
		setarray getelementofarray(getarg(0), .@arr_size++),getarg(.@i);
	}
}

// Begin of the NPC
	.@eac = eaclass();
	.@target = 0;
	if (Can_Change_Fourth()) .@target = roclass((.@eac&~EAJL_UPPER)|EAJL_FOURTH);
	if (.FourthExpanded) {
		switch (.@eac) {
		case EAJ_REBELLION: .@target = roclass(EAJ_NIGHT_WATCH); break;
		case EAJ_KAGEROUOBORO: .@target = roclass(EAJ_SHINKIROSHIRANUI); break;
		case EAJ_SUPER_NOVICE_E: .@target = roclass(EAJ_HYPER_NOVICE); break;
		case EAJ_STAR_EMPEROR: .@target = roclass(EAJ_SKY_EMPEROR); break;
		case EAJ_SOUL_REAPER: .@target = roclass(EAJ_SOUL_ASCETIC); break;
		}
	}
	if (getarg(0) == 0) {
		Doppelganger_Story(.@target);
		close;
	}
	mes .NPCName$;
	if (FourthTrialStage >= 4) {
		mes "You gave a lost soul its freedom. May your new strength always remember that kindness.";
		close;
	}
	mes "You carry the memory of a soul lost between worlds. Tell me what has happened.";
	next;
	mes .NPCName$;
	if (FourthTrialStage == 1 && checkquest(65022) >= 0 && checkquest(65022) != 2) {
		mes "Your search is not yet complete. Speak to Secretary Lime Evenor at Eden Headquarters about the lost soul, then return to me.";
		close;
	}
	if (.@target <= 0 || !.TrialQuest[.@target]) {
		mes "This rite cannot change your present path. Your recorded progress remains safe.";
		close;
	}
	Check_Riding();
	Check_SkillPoints();
	Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
	Confirm_Change(.@target);
	close;

// Displays the job menu
function	Job_Menu	{
	// getarg(0) is the .@job_opt array holding all available job changes.
	function Confirm_Change;
	while(true) {
		.@opt_cnt =  getarraysize(getarg(0));
		if( .@opt_cnt <= 0 ) {
			mes "No more jobs are available.";
			close;
		}

		.@selected = 0; // Just a single job class given, no select needed
		if (.@opt_cnt > 1) {
			// Multiple job classes given. Select one and save it to .@class
			// After that confirm .@class
			mes "Select a job.";
			.@menu$ = "";
			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if( getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FFRebirth^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));
				.@menu$ = .@menu$ + " ~ " + .@jobname$ + ":";
			}
			.@menu$ = .@menu$+" ~ ^777777Cancel^000000";
			.@selected = select(.@menu$) - 1;
			if( .@selected < 0 || .@selected >= .@opt_cnt )
				close;
			next;
			mes .NPCName$;
		}
		.@class = getelementofarray(getarg(0), .@selected);
		if ((.@class == Job_Super_Novice || .@class == Job_Super_Baby) &&
				BaseLevel < .SNovice) {
			// Special Level Requirement because Super Novice and
			// Super Baby can both be selected in one of the first class
			// changes. That's why the Level Requirement is after and not before
			// the selection.
			mes "A base level of " + .SNovice +
				" is required to turn into a " + jobname(.@class) + ".";
			return;
		}
		// Confirm the Class
		Confirm_Change(.@class, .@opt_cnt > 1);
		next;
		mes .NPCName$;
	}
	return;
}


// Preserve prepared souls from older versions while settling unpaid materials at Valkyrie.
function Soul_Is_Prepared {
	.@delivery = (FourthTrialWaived == getarg(0)) ? 65021 : 65023;
	return (checkquest(getarg(0)) == 2 && checkquest(.@delivery) >= 0 && checkquest(.@delivery) != 2);
}

function Soul_Is_Ready {
	return (FourthTrialPaid && Soul_Is_Prepared(getarg(0)));
}

// Valkyrie only: validate first, then commit after the delivery journal is secured.
// Old base payments and old Cardinal waivers each credit their own component.
function Soul_Material_Payment {
	.@holy = getarg(0);
	.@quest = .TrialQuest[FourthTrialJob];
	if (FourthTrialStage != 2 || !.@quest || !fourthtrial_ready(.@quest) ||
		(.@holy && FourthTrialJob != JOB_CARDINAL)) close;
	.@base_due = !FourthTrialPaid;
	.@holy_due = (.@holy && FourthTrialWaived != .@quest);
	.@cost = (.@base_due ? .FourthZenyCost : 0) + (.@holy_due ? .CardinalZenyCost : 0);
	if (Zeny < .@cost || (.@base_due && (countitem(6417) < 100 || countitem(7000) < 200)) ||
		(.@holy_due && countitem(523) < .CardinalWaterCost)) {
		mes "You do not have all the Zeny and materials for this rite. Nothing has been taken; your progress is safe.";
		close;
	}
	if (getarg(1)) {
		if (.@cost) Zeny -= .@cost;
		if (.@base_due) {
			delitem 6417,100;
			delitem 7000,200;
		}
		if (.@holy_due) delitem 523,.CardinalWaterCost;
		FourthTrialPaid = 1;
		if (.@holy) FourthTrialWaived = .@quest;
	}
	return;
}

function Doppelganger_Story {
	.@class = getarg(0);
	.@source = eaclass();
	mes "[Doppelganger]";
	if (FourthTrialStage >= 3) {
		mes "Only a fading echo remains: 'Thank you... The way home is open. Return to Valkyrie.'";
		close;
	}
	if (.@class <= 0 || !.TrialQuest[.@class] || BaseLevel < 200) {
		mes "Your soul follows a different road. The bonds that hold me cannot be broken along your present path.";
		close;
	}
	if (!FourthTrialStage) {
		mes "Do not draw your weapon. I am only what remains of the true Doppelganger. My body is gone, but this fractured dimension will not release my soul.";
		next;
		mes "[Doppelganger]";
		mes "Yet you stand here untouched. Your soul burns with the strength of countless battles... Would you help me find a way out?";
		if (select("I will try to help:Not now") != 1) close;
		next;
		mes "[" + strcharinfo(0) + "]";
		.@reborn = (ADVJOB != 0 || Upper == 1);
		if (.@reborn)
			mes "I cannot free you myself. But I remember someone who guided my soul through rebirth: Valkyrie. Perhaps she can help you too.";
		else
			mes "I have never faced anything like this. But Eden helps adventurers find their way. I will ask Secretary Lime Evenor where to begin.";
		next;
		mes "[Doppelganger]";
		mes (.@reborn ? "Then follow the road you remember, through the Book of Ymir. Please... do not let me become another forgotten shadow." : "Then there is still someone you can ask. Please... tell her a soul is waiting here, caught between worlds.");
		next;
		if (eaclass() != .@source || BaseLevel < 200 || FourthTrialStage) close;
		.@intro_quest = .@reborn ? 65019 : 65022;
		setquest .@intro_quest;
		if (checkquest(.@intro_quest) != 1) {
			mes "The memory could not be recorded. Please speak to me again.";
			close;
		}
		FourthTrialStage = 1;
		mes "[Doppelganger]";
		mes (.@reborn ? "Tell Valkyrie a lost soul is still waiting here. I will hold on until you return." : "You will find Secretary Lime Evenor at Eden Headquarters. I will hold on until you return.");
		close;
	}
	if (FourthTrialStage == 1) {
		mes ((checkquest(65022) >= 0 && checkquest(65022) != 2) ? "Have you spoken to Secretary Lime Evenor at Eden Headquarters? Perhaps she knows who can help a soul trapped between worlds." : "Please ask Valkyrie how to break these bonds. Read the Book of Ymir in Juno's library to reach her hall; ask Metheus Sylphe if you need reading permission.");
		close;
	}
	.@quest = .TrialQuest[.@class];
	if (FourthTrialJob != .@class || !fourthtrial_ready(.@quest) || checkquest(.@quest) < 0) {
		mes "Something in your path has changed. Your progress is preserved; ask Valkyrie before attempting the rite.";
		close;
	}
	if (Soul_Is_Prepared(.@quest) && !FourthTrialPaid) {
		mes "Valkyrie has prepared this light, but the ritual materials are not yet settled. Please return to her first; I cannot complete her work here.";
		close;
	}
	if (!Soul_Is_Ready(.@quest)) {
		mes ((checkquest(.@quest,HUNTING) == 2 || FourthTrialWaived == .@quest) ?
			"You have gathered the souls, but their power is still unbound. Take them to Valkyrie first. Only she can shape them into a soul that can break these bonds." :
			"I can feel the light you have gathered, but it is not yet ready. Collect the assigned boss soul and 10 MVP souls, then bring them to Valkyrie so she can prepare them for this rite.");
		close;
	}
	mes ((FourthTrialWaived == .@quest) ? "That holy soul... Valkyrie has found another way. Its sacred light can break these bonds." : "You came back. Valkyrie has woven those gathered souls into one light... I can feel the chains around me beginning to tremble.");
	next;
	mes "[Doppelganger]";
	mes "Valkyrie has made everything ready. Will you entrust that light to me, so I may finally leave this place?";
	if (select("Release the lost soul:Not yet") != 1) close;
	clear;
	mes "[Doppelganger]";
	if (eaclass() != .@source || FourthTrialStage != 2 || FourthTrialJob != .@class ||
		!fourthtrial_ready(.@quest) || checkquest(.@quest) < 0 ||
		!Soul_Is_Ready(.@quest)) close;
	setquest 65020;
	if (checkquest(65020) != 1) {
		mes "The path back to Valkyrie could not be recorded. Nothing has been taken.";
		close;
	}
	if (checkquest(.@quest) != 2) completequest .@quest;
	if (checkquest(65021) >= 0 && checkquest(65021) != 2) completequest 65021;
	if (checkquest(65023) >= 0 && checkquest(65023) != 2) completequest 65023;
	// Delivery takes no payment. Commit the rescue before cosmetic waits.
	FourthTrialStage = 3;
	mes "At last... I can hear the world beyond this place. The souls you brought are opening the way!";
	specialeffect EF_SOULCHANGE,SELF,"Doppelganger";
	sleep2 800;
	specialeffect EF_POTIONPILLAR,SELF,"Doppelganger";
	sleep2 1000;
	specialeffect EF_LIGHTSPHERE2,SELF,"Doppelganger";
	sleep2 700;
	cloakonnpcself "Doppelganger";
	clear;
	mes "[Doppelganger's fading voice]";
	mes "Thank you, " + strcharinfo(0) + ". You saw a soul where others saw a monster. I will remember your name... wherever this road leads.";
	next;
	mes "[A lingering light]";
	mes "The shadow is gone, but a gentle warmth remains within your soul. Read the Book of Ymir again and tell Valkyrie that he is free.";
	close;
}

// Return each path to its first-job city. Coordinates match the town Warper.
// Called only at the end of the successful awakening ceremony.

	if (FourthTrialStage < 4 || FourthTrialJob != .@job) close;
	switch (.@job) {
	case JOB_DRAGON_KNIGHT:
	case JOB_IMPERIAL_GUARD:
		.@map$ = "izlude"; .@city$ = "Izlude";
		.@x = 128; .@y = (checkre(3) ? 146 : 114); break;
	case JOB_ARCH_MAGE:
	case JOB_ELEMENTAL_MASTER:
		.@map$ = "geffen"; .@city$ = "Geffen";
		.@x = 119; .@y = 59; break;
	case JOB_WINDHAWK:
	case JOB_TROUBADOUR:
	case JOB_TROUVERE:
	case JOB_SKY_EMPEROR:
	case JOB_SOUL_ASCETIC:
		.@map$ = "payon"; .@city$ = "Payon";
		.@x = 179; .@y = 100; break;
	case JOB_CARDINAL:
	case JOB_INQUISITOR:
		.@map$ = "prontera"; .@city$ = "Prontera";
		.@x = 155; .@y = 183; break;
	case JOB_MEISTER:
	case JOB_BIOLO:
		.@map$ = "alberta"; .@city$ = "Alberta";
		.@x = 28; .@y = 234; break;
	case JOB_SHADOW_CROSS:
	case JOB_ABYSS_CHASER:
		.@map$ = "morocc"; .@city$ = "Morroc";
		.@x = 156; .@y = 93; break;
	case JOB_SHINKIRO:
	case JOB_SHIRANUI:
		.@map$ = "amatsu"; .@city$ = "Amatsu";
		.@x = 198; .@y = 84; break;
	case JOB_NIGHT_WATCH:

// Executes the actual jobchange, celebrates, and returns the player to their first-job city.
function	Job_Change	{
	clear;
	.@previous_class = Class;
	.@to_cls = getarg(0);
	.@quest = .TrialQuest[.@to_cls];
	if (!.@quest || !fourthtrial_ready(.@quest) || FourthTrialJob != .@to_cls ||
		FourthTrialStage != 3 || !FourthTrialPaid || checkquest(.@quest) != 2 ||
		checkquest(65020) < 0 || checkquest(65020) == 2) {
		mes .NPCName$;
		mes "Free the lost soul and return to me before receiving this blessing.";
		close;
	}
	Check_Riding();
	Check_SkillPoints();
	Require_Level(.Req_Fourth[0], .Req_Fourth[1]);
	if (.@to_cls == Job_Novice_High && .LastJob)
		lastJob = Class; // Saves the lastJob for rebirth
	jobchange .@to_cls;
	if (Class != .@to_cls) {
		mes .NPCName$;
		mes "The awakening could not finish. Your paid trial is preserved; speak to me again.";
		close;
	}
	completequest 65020;
	FourthTrialStage = 4;
	.@holy_soul = (FourthTrialWaived == .@quest);
	mes .NPCName$;
	mes ((FourthTrialWaived == .@quest) ?
		"The holy light you offered has returned as a blessing. Let it awaken the soul within you. Rise as " :
		"The souls you carried have left their strength within you. I bind that light to your own soul. Rise as ") +
		callfunc("F_InsertArticle", jobname(.@to_cls)) + "!";
	FourthTrialWaived = 0;
	if (.@to_cls == Job_Novice_High)
		resetlvl(1);
	else if (.@to_cls == Job_Baby) {
		resetstatus;
		resetskill;
		set SkillPoint,0;
	}
	if (.@previous_class != Class) {
		if (.Platinum)
			callfunc "F_GetPlatinumSkills";
		if (.GetJobEquip)
			Get_Job_Equip();
	}
	announce "[Ascension] " + strcharinfo(0) + " has ascended to " + jobname(.@to_cls) + "!",bc_all|bc_yellow;
	Awakening_Effects(.@holy_soul);
	Awakening_Departure(.@to_cls);
}

function	Confirm_Change	{
	.@class = getarg(0, -1);
	.@back = getarg(1, false);
	.@source = eaclass();
	if (.@class < 0) {
		mes "Unknown Class Error.";
		close;
	}
	.@quest = .TrialQuest[.@class];
	if (!.@quest || !fourthtrial_ready(.@quest)) {
		mes "This awakening trial is not available. Please contact the administrator.";
		close;
	}
	if (FourthTrialJob && FourthTrialJob != .@class) {
		mes "You have already begun a different awakening path. Your trial has been preserved.";
		close;
	}
	if (FourthTrialStage == 3) {
		mes "The bond is broken. I felt him pass beyond the rift... You have given that lost soul the freedom he could not win alone.";
		next;
		mes .NPCName$;
		mes "You sought no crown for this kindness, yet your soul has grown beyond its former limits. Let me awaken the strength it now carries: ^0055FF" + jobname(.@class) + "^000000.";
		if (select("Accept Valkyrie's blessing:Not yet") != 1) close;
		if (eaclass() != .@source) close;
		Job_Change(.@class);
		close;
	}
	if (checkquest(.@quest) == -1) {
		if (FourthTrialStage != 1 || FourthTrialJob) {
			mes "Your trial record needs attention. No offering has been taken.";
			close;
		}
		mes "A soul imprisoned between worlds... Even one feared as Doppelganger deserves a road home. You were right to bring his plea to me.";
		next;
		mes .NPCName$;
		mes "His chains cannot be broken by steel. Bring me the soul of " + getmonsterinfo(.TrialMob[.@class],MOB_NAME) + ", whose power resonates with your path, and 10 more powerful souls gathered by defeating any MVPs. I will weave their power into a single soul you can carry to him.";
		next;
		mes .NPCName$;
		mes "The first defeat of your assigned boss grants only its special soul. Later defeats may count toward the other 10. Repeated MVP kills count, and you may gather the other souls before or after the assigned one.";
		next;
		mes .NPCName$;
		mes "Accepting this task costs nothing. When you have all the souls, return to me with 100,000,000 Zeny, 100 Silvervine Fruit and 200 Event Stone Coins. The Zeny covers the rare ritual materials I have gathered; your offerings will help bind the souls safely.";
		next;
		if (.@class == JOB_CARDINAL) {

	next;
	mes .NPCName$;
	mes (FourthTrialPaid ? "Your earlier materials payment is fully credited. Shall I weave the souls and entrust their light to you?" : "The materials I gathered cost 100,000,000 Zeny. Bring 100 Silvervine Fruit and 200 Event Stone Coins as well, and I can begin. Once the soul is prepared, there is no payment at Doppelganger.");

	clear;
	mes .NPCName$;

		!fourthtrial_ready(.@quest) || checkquest(.@quest,HUNTING) != 2 ||
		FourthTrialWaived == .@quest || Soul_Is_Ready(.@quest)) close;

	if (checkquest(65023) < 0 || checkquest(65023) == 2) {

	mes .NPCName$;
	mes "The woven soul has passed into your care, and the ritual is fully paid. Take it to Doppelganger. Once he is free, return to me; there is still a light in you waiting to awaken.";
	close;
}

		if (BaseLevel < 200 || ADVJOB != 0 || Upper == 1 || valkyrie_Q == 0) close;

		close2;
