//===== Ragnarok Offline =====================================
//= Card Collector
//===== Description: =========================================
//= Buys standalone card items for character-local Black Market Points.
//= Offers scale with the canonical source monster level:
//= Normal 1-50: 250-1000; 51-100: 2000-4000; 101+: 7000-15000;
//= Miniboss: 7000-20000; MVP: 20000-100000.
//= Account-wide loyalty counters reward repeat Normal, Miniboss,
//= and MVP card sales. Multiple milestones roll independently.
//= 1.1 Look up Normal cards from card_seller_creation at click time
//=     so OnInit hash order cannot drop the Normal catalog.
//============================================================

paramk,41,159,4	script	Black Market Boss#offline	4_M_MANAGER,{
	mes "[Black Market Boss]";
	mes "So, you brought me cards.";
	mes "Good. Every card that changes hands in this market eventually reaches my table.";
	mes "I pay in ^0055FFBlack Market Points^000000, not Zeny.";
	mes "My offer depends on the monster behind each card.";
	next;
	switch (select("Sell one card type.:Sell every accepted card.:Check points and loyalty.:Leave.")) {
	case 1:
		clear;
		goto L_Single;
	case 2:
		clear;
		goto L_Bulk;
	case 3:
		clear;
		mes "[Black Market Boss]";
		mes "You have ^0055FF" + BlackMarketPoints + " Black Market Points^000000.";
		mes "Normal-card loyalty: ^0055FF" + #CC_Normal + " / 30^000000.";
		mes "Miniboss-card loyalty: ^0055FF" + #CC_Mini + " / 5^000000.";
		mes "MVP-card loyalty: ^0055FF" + #CC_MVP + " / 5^000000.";
		close;

	default:
		close;
	}

L_Single:
		getinventorylist;
		.@card_count = 0;
		for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
			.@item_id = @inventorylist_id[.@i];
			if (getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_CARD ||
				getiteminfo(.@item_id, ITEMINFO_SUBTYPE) != CARD_NORMAL ||
				callsub(S_GetOffer, .@item_id) <= 0)
				continue;
			if (inarray(.@card_ids[0], .@item_id) == -1)
				.@card_ids[.@card_count++] = .@item_id;
		}

		if (!.@card_count) {
			mes "[Black Market Boss]";
			mes "Do not waste my time.";
			mes "You are not carrying any cards I accept.";
			mes "Cards inserted into equipment don't count.";
			close;
		}

		.@page = 0;
		while (true) {
			.@start = .@page * .PageSize;
			.@end = min(.@start + .PageSize, .@card_count);

			mes "[Black Market Boss]";
			mes "Put a card on the table.";
			mes "^777777Page " + (.@page + 1) + " / " + ((.@card_count + .PageSize - 1) / .PageSize) + "^000000";
			for (.@i = .@start; .@i < .@end; ++.@i)
				mes "^0055FF" + (.@i + 1) + ".^000000 " + mesitemlink(.@card_ids[.@i], false) + " x" + countitem(.@card_ids[.@i]) + " (" + callsub(S_GetOffer, .@card_ids[.@i]) + " CP each)";
			next;

			.@menu$ = "Choose a card.";
			.@option_count = 1;
			.@next_option = 0;
			.@previous_option = 0;
			if (.@end < .@card_count) {
				.@menu$ += ":Next page.";
				.@next_option = ++.@option_count;
			}
			if (.@page > 0) {
				.@menu$ += ":Previous page.";
				.@previous_option = ++.@option_count;
			}
			.@menu$ += ":Cancel.";
			.@cancel_option = ++.@option_count;

			.@action = select(.@menu$);
			clear;
			if (.@action == .@next_option) {
				++.@page;
				continue;
			}
			if (.@action == .@previous_option) {
				--.@page;
				continue;
			}
			if (.@action == .@cancel_option)
				close;
			break;
		}

		mes "[Black Market Boss]";
		mes "Enter the number shown beside the card.";
		input .@selection, .@start + 1, .@end;
		clear;
		.@selected_id = .@card_ids[.@selection - 1];
		.@owned = countitem(.@selected_id);

		mes "[Black Market Boss]";
		mes "You have " + .@owned + " " + mesitemlink(.@selected_id, false) + ".";
		mes "How many are you offering me?";
		input .@amount, 1, .@owned;
		clear;
		.@points_each = callsub(S_GetOffer, .@selected_id);
		.@payment = .@amount * .@points_each;
		.@tier = callsub(S_GetTier, .@selected_id);

		mes "[Black Market Boss]";
		mes "^0055FF" + mesitemlink(.@selected_id, false) + "^000000 x" + .@amount;
		mes "My offer: ^0055FF" + .@payment + " Black Market Points^000000.";
		mes "Once they enter my collection, the deal is final.";
		next;
		if (select("Sell the cards.:Keep them.") != 1)
			close;
		clear;

		// Roll every completed loyalty milestone before committing so the
		// complete reward bundle can be checked for inventory capacity.
		switch (.@tier) {
		case 1:
			.@new_progress = #CC_Normal + .@amount;
			.@milestones = .@new_progress / 30;
			.@remainder = .@new_progress % 30;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				if (rand(100) < 70)
					.@old_card_albums += 5;
				else
					.@dead_branches += 5;
			}
			break;
		case 2:
			.@new_progress = #CC_Mini + .@amount;
			.@milestones = .@new_progress / 5;
			.@remainder = .@new_progress % 5;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				if (rand(100) < 50)
					.@magic_card_albums += 1;
				else
					.@dead_branches += 10;
			}
			break;
		case 3:
			.@new_progress = #CC_MVP + .@amount;
			.@milestones = .@new_progress / 5;
			.@remainder = .@new_progress % 5;
			for (.@i = 0; .@i < .@milestones; ++.@i) {
				// 1% album, 40% Bloody Branch x2, 40% Empty Bottle x100, 19% backstab.
				.@roll = rand(100);
				if (.@roll < 1)
					.@mvp_card_albums += 1;
				else if (.@roll < 41)
					.@bloody_branches += 2;
				else if (.@roll < 81)
					.@empty_bottles += 100;
				else
					.@backstabs += 1;
			}
			break;
		case 4:
			// Source-less and Special cards pay Black Market Points only. They do
			// not participate in the monster-card loyalty promotions.
			.@milestones = 0;
			break;
		default:
			mes "[Black Market Boss]";
			mes "That card no longer belongs in my catalog.";
			mes "Take it away.";
			close;
		}

		.@reward_count = 0;
		if (.@old_card_albums) {
			.@reward_ids[.@reward_count] = 616;
			.@reward_amounts[.@reward_count++] = .@old_card_albums;
		}
		if (.@dead_branches) {
			.@reward_ids[.@reward_count] = 604;
			.@reward_amounts[.@reward_count++] = .@dead_branches;
		}
		if (.@magic_card_albums) {
			.@reward_ids[.@reward_count] = 12246;
			.@reward_amounts[.@reward_count++] = .@magic_card_albums;
		}
		if (.@mvp_card_albums) {
			.@reward_ids[.@reward_count] = 35000;
			.@reward_amounts[.@reward_count++] = .@mvp_card_albums;
		}
		if (.@bloody_branches) {
			.@reward_ids[.@reward_count] = 12103;
			.@reward_amounts[.@reward_count++] = .@bloody_branches;
		}
		if (.@empty_bottles) {
			.@reward_ids[.@reward_count] = 713;
			.@reward_amounts[.@reward_count++] = .@empty_bottles;
		}

		if (.@reward_count && !checkweight2(.@reward_ids, .@reward_amounts)) {
			mes "[Black Market Boss]";
			mes "My payment includes more than your pockets can hold.";
			mes "Make room, then return to my table.";
			close;
		}

		// Revalidate immediately before committing the exchange.
		if (getiteminfo(.@selected_id, ITEMINFO_TYPE) != IT_CARD ||
			getiteminfo(.@selected_id, ITEMINFO_SUBTYPE) != CARD_NORMAL ||
			callsub(S_GetOffer, .@selected_id) != .@points_each ||
			callsub(S_GetTier, .@selected_id) != .@tier ||
			countitem(.@selected_id) < .@amount) {
			mes "[Black Market Boss]";
			mes "Something changed before the deal was closed.";
			mes "I do not conduct uncertain business.";
			close;
		}

		delitem .@selected_id, .@amount;
		BlackMarketPoints += .@payment;
		switch (.@tier) {
		case 1: #CC_Normal = .@remainder; break;
		case 2: #CC_Mini = .@remainder; break;
		case 3: #CC_MVP = .@remainder; break;
		}
		for (.@i = 0; .@i < .@reward_count; ++.@i)
			getitem .@reward_ids[.@i], .@reward_amounts[.@i];
		logmes "Card Collector: sold " + .@amount + " x item " + .@selected_id + " for " + .@payment + " Black Market Points; tier " + .@tier + "; milestones " + .@milestones + "; OCA " + .@old_card_albums + "; DB " + .@dead_branches + "; MCA " + .@magic_card_albums + "; MVP albums " + .@mvp_card_albums + "; BDB " + .@bloody_branches + "; bottles " + .@empty_bottles + "; backstabs " + .@backstabs + "; balance " + BlackMarketPoints;

		mes "[Black Market Boss]";
		mes "A clean transaction. I expect nothing less.";
		mes "You received ^0055FF" + .@payment + " Black Market Points^000000.";
		mes "Current balance: ^0055FF" + BlackMarketPoints + " Black Market Points^000000.";
		if (.@milestones) {
			mes "^FF8800Loyalty reward:^000000";
			if (.@old_card_albums) mes mesitemlink(616, false) + " x" + .@old_card_albums;
			if (.@dead_branches) mes mesitemlink(604, false) + " x" + .@dead_branches;
			if (.@magic_card_albums) mes mesitemlink(12246, false) + " x" + .@magic_card_albums;
			if (.@mvp_card_albums) mes mesitemlink(35000, false) + " x" + .@mvp_card_albums;
			if (.@bloody_branches) mes mesitemlink(12103, false) + " x" + .@bloody_branches;
			if (.@empty_bottles) mes mesitemlink(713, false) + " x" + .@empty_bottles;
		}
		if (.@backstabs) {
			mes "...";
			next;
			mes "[Black Market Boss]";
			mes "You sold the wrong card to the wrong man.";
			mes "A collector should know when not to sell.";
			next;
			close2;
			unitkill getcharid(3);
			end;
		}
		close;

L_Bulk:
	getinventorylist;
	deletearray .@bulk_ids[0], getarraysize(.@bulk_ids);
	deletearray .@bulk_amounts[0], getarraysize(.@bulk_amounts);
	deletearray .@bulk_offers[0], getarraysize(.@bulk_offers);
	deletearray .@bulk_tiers[0], getarraysize(.@bulk_tiers);
	.@bulk_count = 0;
	.@bulk_cards = 0;
	.@bulk_payment = 0;
	.@bulk_normal = 0;
	.@bulk_mini = 0;
	.@bulk_mvp = 0;
	.@bulk_special = 0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		.@item_id = @inventorylist_id[.@i];
		if (inarray(.@bulk_ids[0], .@item_id) != -1 ||
			getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_CARD ||
			getiteminfo(.@item_id, ITEMINFO_SUBTYPE) != CARD_NORMAL)
			continue;
		.@offer = callsub(S_GetOffer, .@item_id);
		.@tier = callsub(S_GetTier, .@item_id);
		if (.@offer <= 0 || .@tier < 1 || .@tier > 4)
			continue;
		.@amount = countitem(.@item_id);
		if (.@amount < 1)
			continue;
		.@bulk_ids[.@bulk_count] = .@item_id;
		.@bulk_amounts[.@bulk_count] = .@amount;
		.@bulk_offers[.@bulk_count] = .@offer;
		.@bulk_tiers[.@bulk_count++] = .@tier;
		.@bulk_cards += .@amount;
		.@bulk_payment += .@amount * .@offer;
		switch (.@tier) {
		case 1: .@bulk_normal += .@amount; break;
		case 2: .@bulk_mini += .@amount; break;
		case 3: .@bulk_mvp += .@amount; break;
		case 4: .@bulk_special += .@amount; break;
		}
	}

	if (!.@bulk_count) {
		mes "[Black Market Boss]";
		mes "Do not waste my time.";
		mes "You are not carrying any cards I accept.";
		mes "Cards inserted into equipment don't count.";
		close;
	}

	mes "[Black Market Boss]";
	mes "You are offering every card I accept.";
	mes "^0055FF" + .@bulk_count + " card types, " + .@bulk_cards + " cards total.^000000";
	mes "Normal: " + .@bulk_normal + "  Miniboss: " + .@bulk_mini + "  MVP: " + .@bulk_mvp;
	mes "Special: " + .@bulk_special + " ^777777(no loyalty)^000000";
	mes "My offer: ^0055FF" + .@bulk_payment + " Black Market Points^000000.";
	mes "Once they enter my collection, the deal is final.";
	next;
	if (select("Sell every accepted card.:Keep everything.") != 1)
		close;
	clear;

	.@new_normal = #CC_Normal + .@bulk_normal;
	.@normal_milestones = .@new_normal / 30;
	.@normal_remainder = .@new_normal % 30;
	for (.@i = 0; .@i < .@normal_milestones; ++.@i) {
		if (rand(100) < 70)
			.@old_card_albums += 5;
		else
			.@dead_branches += 5;
	}
	.@new_mini = #CC_Mini + .@bulk_mini;
	.@mini_milestones = .@new_mini / 5;
	.@mini_remainder = .@new_mini % 5;
	for (.@i = 0; .@i < .@mini_milestones; ++.@i) {
		if (rand(100) < 50)
			.@magic_card_albums += 1;
		else
			.@dead_branches += 10;
	}
	.@new_mvp = #CC_MVP + .@bulk_mvp;
	.@mvp_milestones = .@new_mvp / 5;
	.@mvp_remainder = .@new_mvp % 5;
	for (.@i = 0; .@i < .@mvp_milestones; ++.@i) {
		.@roll = rand(100);
		if (.@roll < 1)
			.@mvp_card_albums += 1;
		else if (.@roll < 41)
			.@bloody_branches += 2;
		else if (.@roll < 81)
			.@empty_bottles += 100;
		else
			.@backstabs += 1;
	}

	.@reward_count = 0;
	if (.@old_card_albums) { .@reward_ids[.@reward_count] = 616; .@reward_amounts[.@reward_count++] = .@old_card_albums; }
	if (.@dead_branches) { .@reward_ids[.@reward_count] = 604; .@reward_amounts[.@reward_count++] = .@dead_branches; }
	if (.@magic_card_albums) { .@reward_ids[.@reward_count] = 12246; .@reward_amounts[.@reward_count++] = .@magic_card_albums; }
	if (.@mvp_card_albums) { .@reward_ids[.@reward_count] = 35000; .@reward_amounts[.@reward_count++] = .@mvp_card_albums; }
	if (.@bloody_branches) { .@reward_ids[.@reward_count] = 12103; .@reward_amounts[.@reward_count++] = .@bloody_branches; }
	if (.@empty_bottles) { .@reward_ids[.@reward_count] = 713; .@reward_amounts[.@reward_count++] = .@empty_bottles; }
	if (.@reward_count && !checkweight2(.@reward_ids, .@reward_amounts)) {
		mes "[Black Market Boss]";
		mes "My payment includes more than your pockets can hold.";
		mes "Make room, then return to my table.";
		close;
	}

	for (.@i = 0; .@i < .@bulk_count; ++.@i) {
		if (getiteminfo(.@bulk_ids[.@i], ITEMINFO_TYPE) != IT_CARD ||
			getiteminfo(.@bulk_ids[.@i], ITEMINFO_SUBTYPE) != CARD_NORMAL ||
			callsub(S_GetOffer, .@bulk_ids[.@i]) != .@bulk_offers[.@i] ||
			callsub(S_GetTier, .@bulk_ids[.@i]) != .@bulk_tiers[.@i] ||
			countitem(.@bulk_ids[.@i]) < .@bulk_amounts[.@i]) {
			mes "[Black Market Boss]";
			mes "Something changed before the deal was closed.";
			mes "I do not conduct uncertain business.";
			close;
		}
	}
	for (.@i = 0; .@i < .@bulk_count; ++.@i)
		delitem .@bulk_ids[.@i], .@bulk_amounts[.@i];
	BlackMarketPoints += .@bulk_payment;
	#CC_Normal = .@normal_remainder;
	#CC_Mini = .@mini_remainder;
	#CC_MVP = .@mvp_remainder;
	for (.@i = 0; .@i < .@reward_count; ++.@i)
		getitem .@reward_ids[.@i], .@reward_amounts[.@i];
	logmes "Card Collector: bulk sold " + .@bulk_cards + " cards across " + .@bulk_count + " item IDs for " + .@bulk_payment + " Black Market Points; Normal " + .@bulk_normal + " (" + .@normal_milestones + " milestones); Miniboss " + .@bulk_mini + " (" + .@mini_milestones + " milestones); MVP " + .@bulk_mvp + " (" + .@mvp_milestones + " milestones); Special " + .@bulk_special + " (no loyalty); OCA " + .@old_card_albums + "; DB " + .@dead_branches + "; MCA " + .@magic_card_albums + "; MVP albums " + .@mvp_card_albums + "; BDB " + .@bloody_branches + "; bottles " + .@empty_bottles + "; backstabs " + .@backstabs + "; balance " + BlackMarketPoints;

	mes "[Black Market Boss]";
	mes "A clean transaction. I expect nothing less.";
	mes "You received ^0055FF" + .@bulk_payment + " Black Market Points^000000.";
	mes "Current balance: ^0055FF" + BlackMarketPoints + " Black Market Points^000000.";
	if (.@reward_count) {
		mes "^FF8800Loyalty reward:^000000";
		if (.@old_card_albums) mes mesitemlink(616, false) + " x" + .@old_card_albums;
		if (.@dead_branches) mes mesitemlink(604, false) + " x" + .@dead_branches;
		if (.@magic_card_albums) mes mesitemlink(12246, false) + " x" + .@magic_card_albums;
		if (.@mvp_card_albums) mes mesitemlink(35000, false) + " x" + .@mvp_card_albums;
		if (.@bloody_branches) mes mesitemlink(12103, false) + " x" + .@bloody_branches;
		if (.@empty_bottles) mes mesitemlink(713, false) + " x" + .@empty_bottles;
	}
	if (.@backstabs) {
		mes "...";
		next;
		mes "[Black Market Boss]";
		mes "You sold the wrong card to the wrong man.";
		mes "A collector should know when not to sell.";
		next;
		close2;
		unitkill getcharid(3);
		end;
	}
	close;

S_GetOffer:
	.@item_id = getarg(0);
	if (.@item_id == 6716 || .@item_id == 6717 || .@item_id == 6718)
		return 300;
	return bmc_cardoffer(.@item_id);

S_GetTier:
	.@item_id = getarg(0);
	if (.@item_id == 6716 || .@item_id == 6717 || .@item_id == 6718)
		return 1;
	return bmc_cardtier(.@item_id);

OnInit:
	.PageSize = 10;

	setunittitle(getnpcid(0), "<Card Collector>");
	end;
}

function	script	F_OfflineMvpCardAlbum	{
	.@size = bmc_mvpcount();
	if (.@size <= 0) {
		message strcharinfo(0), "The MVP Card Album is not ready yet.";
		getitem 35000,1;
		return;
	}
	getitem bmc_mvpid(rand(.@size)),1;
	return;
}

// Bodyguards: visible and proximity-reactive, but they do not open dialogue.
paramk,41,161,2	script	Guard#card_collector_1	405,2,1,{
	callsub S_Warn;
	end;

OnTouch:
	callsub S_Warn;
	end;

S_Warn:
	if (@CardGuardWarningUntil > gettimetick(2))
		return;
	@CardGuardWarningUntil = gettimetick(2) + 3;

	switch (rand(3)) {
	case 0:
		npctalk "Easy. Don't try anything.", strnpcinfo(0), bc_self;
		break;
	case 1:
		npctalk "Watch yourself. The Boss is protected.", strnpcinfo(0), bc_self;
		break;
	default:
		npctalk "Keep your hands where I can see them.", strnpcinfo(0), bc_self;
		break;
	}
	return;
}

// Room-entrance guard: same balloon warning pattern, different sprite and lines.
paramk,14,175,4	script	Guard#black_market_door	686,2,1,{
	mes "[Guard]";
	mes "This isn't a hallway. State your business.";
	next;
	if (select("How are the market rates today?:Never mind.") == 2)
		close;
	clear;

	mes "[Guard]";
	mes "The current adjustment is against each drawer's listed Black Market Point price.";
	mes "Harlan: " + $BlackMarketRateHarlan + "%";
	mes "Soren: " + $BlackMarketRateSoren + "%";
	mes "Mordain: " + $BlackMarketRateMordain + "%";
	mes "Kaedra: " + $BlackMarketRateKaedra + "%";
	mes "Tess: " + $BlackMarketRateTess + "%";
	mes "Weaver: " + $BlackMarketRateWeaver + "%";
	mes "Lady Seraphine: " + $BlackMarketRateSeraphine + "%";
mes "Madame Celestine: " + $BlackMarketRateSpecialty + "%";
	mes "Cards (Corvin, Corven, Veynar): " + $BlackMarketRateCards + "%";
	close;

OnTouch:
	callsub S_Warn;
	end;

S_Warn:
	if (@BmDoorGuardWarningUntil > gettimetick(2))
		return;

	switch (rand(3)) {
	case 0:
		npctalk "This isn't a hallway. State your business or move.", strnpcinfo(0), bc_self;
		break;
	case 1:
		npctalk "Don't stand in the doorway. People inside don't like gawkers.", strnpcinfo(0), bc_self;
		break;
	default:
		npctalk "Keep your eyes down and your mouth shut.", strnpcinfo(0), bc_self;

	}
	return;
}
