//===== Ragnarok Offline =====================================
//= Treasure Collector
//===== Description: =========================================
//= Buys the 21 project Treasure Boxes for fixed Black Market Points.
//= Iron, Silver, and Gold share 30-box loyalty milestones.
//= Diamond and Platinum use five-box milestones.
//= Platinum milestones have an 81% Platinum III reward and 19% backstab.
//============================================================

paramk,41,159,4	script	Black Market Boss#offline	4_M_MANAGER,{
	mes "[Black Market Boss]";
	mes "So, you brought me treasure.";
	next;
	mes "[Black Market Boss]";
	mes "Cards stay with their dealers. Treasure comes to my table.";
	next;
	mes "[Black Market Boss]";
	mes "I pay in ^0055FFBlack Market Points^000000.";
	next;
	switch (select("Sell one treasure type.:Sell every accepted treasure.:Check points and loyalty.:Leave.")) {
	case 1:
		clear;
		goto L_Single;
	case 2:
		clear;
		goto L_Bulk;
	case 3:
		clear;
		mes "[Black Market Boss]";
		mes "You have ^0055FF" + BlackMarketPoints + " Black Market Points^000000.";
		next;
		mes "[Black Market Boss]";
		mes "Iron, Silver, and Gold loyalty: ^0055FF" + #CC_Normal + " / 30^000000.";
		next;
		mes "[Black Market Boss]";
		mes "Diamond loyalty: ^0055FF" + #CC_Mini + " / 5^000000.";
		next;
		mes "[Black Market Boss]";
		mes "Platinum loyalty: ^0055FF" + #CC_MVP + " / 5^000000.";
		next;
		mes "[Black Market Boss]";
		mes "Normal rewards advance Iron to Silver I, Silver to Gold I, or Gold to Diamond I.";
		next;
		mes "[Black Market Boss]";
		mes "Diamond rewards advance to Platinum I. Platinum rewards can yield Platinum III.";
		close;
	default:
		close;
	}

L_Single:
	.@sale_count = 0;
	for (.@i = 0; .@i < 21; ++.@i) {
		.@item_id = 9000001 + .@i;
		if (countitem(.@item_id) <= 0 || callsub(S_GetOffer, .@item_id) <= 0)
			continue;
		.@sale_ids[.@sale_count] = .@item_id;
		.@sale_amounts[.@sale_count] = countitem(.@item_id);
		.@sale_offers[.@sale_count] = callsub(S_GetOffer, .@item_id);
		.@sale_tiers[.@sale_count] = callsub(S_GetTier, .@item_id);
		.@sale_families[.@sale_count++] = callsub(S_GetFamily, .@item_id);
	}
	if (!.@sale_count) {
		mes "[Black Market Boss]";
		mes "You are not carrying any treasure I accept.";
		close;
	}
	.@page = 0;
	while (true) {
		.@start = .@page * .PageSize;
		.@end = min(.@start + .PageSize, .@sale_count);
		.@menu$ = "Cancel.";
		for (.@i = .@start; .@i < .@end; ++.@i)
			.@menu$ += ":" + getitemname(.@sale_ids[.@i]) + " x" + .@sale_amounts[.@i] + " (" + .@sale_offers[.@i] + " BP each)";
		.@next_option = 0;
		.@previous_option = 0;
		if (.@end < .@sale_count) {
			.@menu$ += ":Next page.";
			.@next_option = .@end - .@start + 2;
		}
		if (.@page > 0) {
			.@menu$ += ":Previous page.";
			.@previous_option = .@end - .@start + 2 + (.@end < .@sale_count);
		}
		mes "[Black Market Boss]";
		mes "Choose a treasure type. Each option shows your quantity and my offer.";
		.@action = select(.@menu$);
		if (.@action == 1)
			close;
		if (.@action == .@next_option) {
			++.@page;
			clear;
			continue;
		}
		if (.@action == .@previous_option) {
			--.@page;
			clear;
			continue;
		}
		.@selected = .@start + .@action - 2;
		break;
	}
	clear;
	mes "[Black Market Boss]";
	mes "How many " + getitemname(.@sale_ids[.@selected]) + " are you offering me?";
	.@input_result = input(.@amount, 1, .@sale_amounts[.@selected]);
	if (.@input_result != 0)
		close;
	.@selected_id = .@sale_ids[.@selected];
	.@selected_offer = .@sale_offers[.@selected];
	.@selected_tier = .@sale_tiers[.@selected];
	.@selected_family = .@sale_families[.@selected];
	.@sale_ids[0] = .@selected_id;
	.@sale_amounts[0] = .@amount;
	.@sale_offers[0] = .@selected_offer;
	.@sale_tiers[0] = .@selected_tier;
	.@sale_families[0] = .@selected_family;
	.@sale_count = 1;
	L_Review:
	.@sale_total = 0;
	.@sale_normal = 0;
	.@sale_diamond = 0;
	.@sale_platinum = 0;
	.@highest_normal_family = 0;
	for (.@i = 0; .@i < .@sale_count; ++.@i) {
		.@sale_total += .@sale_amounts[.@i] * .@sale_offers[.@i];
		switch (.@sale_tiers[.@i]) {
		case 1:
			.@sale_normal += .@sale_amounts[.@i];
			.@highest_normal_family = max(.@highest_normal_family, .@sale_families[.@i]);
			break;
		case 2: .@sale_diamond += .@sale_amounts[.@i]; break;
		case 3: .@sale_platinum += .@sale_amounts[.@i]; break;
		}
	}
	.@normal_before = #CC_Normal;
	.@diamond_before = #CC_Mini;
	.@platinum_before = #CC_MVP;
	.@normal_total = .@normal_before + .@sale_normal;
	.@diamond_total = .@diamond_before + .@sale_diamond;
	.@platinum_total = .@platinum_before + .@sale_platinum;
	.@normal_milestones = .@normal_total / 30;
	.@diamond_milestones = .@diamond_total / 5;
	.@platinum_milestones = .@platinum_total / 5;
	.@normal_remainder = .@normal_total % 30;
	.@diamond_remainder = .@diamond_total % 5;
	.@platinum_remainder = .@platinum_total % 5;
	.@reward_count = 0;
	.@backstabs = 0;
	for (.@i = 0; .@i < .@normal_milestones; ++.@i) {
		.@reward_id = 9000001;
		if (.@highest_normal_family == 1) .@reward_id = 9000006;
		else if (.@highest_normal_family == 2) .@reward_id = 9000011;
		else if (.@highest_normal_family == 3) .@reward_id = 9000016;
		.@reward_ids[.@reward_id - 9000001] += 1;
	}
	for (.@i = 0; .@i < .@diamond_milestones; ++.@i)
		.@reward_ids[18] += 1;
	clear;
	for (.@i = 0; .@i < .@sale_count; ++.@i) {
		mes "[Black Market Boss]";
		mes "Offering: " + getitemname(.@sale_ids[.@i]) + " x" + .@sale_amounts[.@i] + " (" + .@sale_offers[.@i] + " BP each).";
		next;
	}
	mes "[Black Market Boss]";
	mes "Iron/Silver/Gold: " + .@sale_normal + "; Diamond: " + .@sale_diamond + "; Platinum: " + .@sale_platinum + ".";
	next;
	mes "[Black Market Boss]";
	mes "My total offer is ^0055FF" + .@sale_total + " Black Market Points^000000.";
	next;
	mes "[Black Market Boss]";
	mes "Once they enter my collection, the deal is final.";
	next;
	if (select("Sell the treasure.:Keep it.") != 1)
		close;
	clear;
	for (.@i = 0; .@i < .@platinum_milestones; ++.@i) {
		if (rand(100) < 81)
			.@reward_ids[20] += 1;
		else
			++.@backstabs;
	}
	deletearray .@reward_list_ids[0], getarraysize(.@reward_list_ids);
	deletearray .@reward_list_amounts[0], getarraysize(.@reward_list_amounts);
	.@reward_count = 0;
	for (.@i = 0; .@i < 21; ++.@i) {
		if (.@reward_ids[.@i] <= 0)
			continue;
		.@reward_list_ids[.@reward_count] = 9000001 + .@i;
		.@reward_list_amounts[.@reward_count++] = .@reward_ids[.@i];
	}

	if (.@reward_count && !checkweight2(.@reward_list_ids, .@reward_list_amounts)) {
		mes "[Black Market Boss]";
		mes "My reward will not fit in your pockets. Make room and return.";
		close;
	}
	if (#CC_Normal != .@normal_before || #CC_Mini != .@diamond_before || #CC_MVP != .@platinum_before) {
		mes "[Black Market Boss]";
		mes "Something changed before the deal was closed. I do not conduct uncertain business.";
		close;
	}
	for (.@i = 0; .@i < .@sale_count; ++.@i) {
		if (getiteminfo(.@sale_ids[.@i], ITEMINFO_TYPE) != IT_ETC ||
			callsub(S_GetOffer, .@sale_ids[.@i]) != .@sale_offers[.@i] ||
			callsub(S_GetTier, .@sale_ids[.@i]) != .@sale_tiers[.@i] ||
			countitem(.@sale_ids[.@i]) < .@sale_amounts[.@i]) {
			mes "[Black Market Boss]";
			mes "Something changed before the deal was closed. I do not conduct uncertain business.";
			close;
		}
	}
	for (.@i = 0; .@i < .@sale_count; ++.@i)
		delitem .@sale_ids[.@i], .@sale_amounts[.@i];
	BlackMarketPoints += .@sale_total;
	#CC_Normal = .@normal_remainder;
	#CC_Mini = .@diamond_remainder;
	#CC_MVP = .@platinum_remainder;
	for (.@i = 0; .@i < .@reward_count; ++.@i)
		getitem .@reward_list_ids[.@i], .@reward_list_amounts[.@i];
	.@log$ = "Treasure Collector: sold";
	for (.@i = 0; .@i < .@sale_count; ++.@i)
		.@log$ += " " + .@sale_ids[.@i] + "x" + .@sale_amounts[.@i];
	.@log$ += " for " + .@sale_total + " Black Market Points; highest Normal family " + .@highest_normal_family + "; milestones " + .@normal_milestones + "/" + .@diamond_milestones + "/" + .@platinum_milestones + "; remainders " + .@normal_remainder + "/" + .@diamond_remainder + "/" + .@platinum_remainder + "; backstabs " + .@backstabs + "; balance " + BlackMarketPoints;
	for (.@i = 0; .@i < .@reward_count; ++.@i)
		.@log$ += "; reward " + .@reward_list_ids[.@i] + "x" + .@reward_list_amounts[.@i];
	logmes .@log$;

	mes "[Black Market Boss]";
	mes "A clean transaction. I expect nothing less.";
	next;
	mes "[Black Market Boss]";
	mes "You received ^0055FF" + .@sale_total + " Black Market Points^000000.";
	next;
	mes "[Black Market Boss]";
	mes "Current balance: ^0055FF" + BlackMarketPoints + " Black Market Points^000000.";
	next;
	for (.@i = 0; .@i < .@reward_count; ++.@i) {
		mes "[Black Market Boss]";
		mes "Loyalty reward: " + mesitemlink(.@reward_list_ids[.@i], false) + " x" + .@reward_list_amounts[.@i] + ".";
		next;
	}
	if (.@backstabs) {
		mes "[Black Market Boss]";
		mes "You should have known when not to sell.";
		next;
		close2;
		specialeffect2 EF_BACKSTAP;
		unitkill getcharid(3);
		end;
	}
	close;

L_Bulk:
	.@sale_count = 0;
	for (.@i = 0; .@i < 21; ++.@i) {
		.@item_id = 9000001 + .@i;
		.@amount = countitem(.@item_id);
		if (.@amount <= 0 || callsub(S_GetOffer, .@item_id) <= 0)
			continue;
		.@sale_ids[.@sale_count] = .@item_id;
		.@sale_amounts[.@sale_count] = .@amount;
		.@sale_offers[.@sale_count] = callsub(S_GetOffer, .@item_id);
		.@sale_tiers[.@sale_count] = callsub(S_GetTier, .@item_id);
		.@sale_families[.@sale_count++] = callsub(S_GetFamily, .@item_id);
	}
	if (!.@sale_count) {
		mes "[Black Market Boss]";
		mes "You are not carrying any treasure I accept.";
		close;
	}
	.@page = 0;
	while (true) {
		.@start = .@page * .PageSize;
		.@end = min(.@start + .PageSize, .@sale_count);
		clear;
		for (.@i = .@start; .@i < .@end; ++.@i) {
			mes "[Black Market Boss]";
			mes getitemname(.@sale_ids[.@i]) + " x" + .@sale_amounts[.@i] + " (" + .@sale_offers[.@i] + " BP each).";
			next;
		}
		.@menu$ = "Review total.";
		.@review_option = 1;
		.@next_option = 0;
		.@previous_option = 0;
		if (.@end < .@sale_count) {
			.@menu$ += ":Next page.";
			.@next_option = 2;
		}
		if (.@page > 0) {
			.@menu$ += ":Previous page.";
			.@previous_option = .@next_option ? 3 : 2;
		}
		.@menu$ += ":Cancel.";
		.@cancel_option = select(.@menu$);
		if (.@cancel_option == .@review_option)
			break;
		if (.@cancel_option == .@next_option) {
			++.@page;
			continue;
		}
		if (.@cancel_option == .@previous_option) {
			--.@page;
			continue;
		}
		close;
	}
	goto L_Review;

S_GetOffer:
	.@item_id = getarg(0);
	if (.@item_id < 9000001 || .@item_id > 9000021 || getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_ETC)
		return 0;
	return .TreasureOffers[.@item_id - 9000001];

S_GetTier:
	.@item_id = getarg(0);
	if (.@item_id >= 9000001 && .@item_id <= 9000015) return 1;
	if (.@item_id >= 9000016 && .@item_id <= 9000018) return 2;
	if (.@item_id >= 9000019 && .@item_id <= 9000021) return 3;
	return 0;

S_GetFamily:
	.@item_id = getarg(0);
	if (.@item_id >= 9000001 && .@item_id <= 9000005) return 1;
	if (.@item_id >= 9000006 && .@item_id <= 9000010) return 2;
	if (.@item_id >= 9000011 && .@item_id <= 9000015) return 3;
	if (.@item_id >= 9000016 && .@item_id <= 9000018) return 4;
	if (.@item_id >= 9000019 && .@item_id <= 9000021) return 5;
	return 0;

OnInit:
	.PageSize = 10;
	.TreasureOffers[0] = 360;
	.TreasureOffers[1] = 400;
	.TreasureOffers[2] = 600;
	.TreasureOffers[3] = 700;
	.TreasureOffers[4] = 800;
	.TreasureOffers[5] = 1400;
	.TreasureOffers[6] = 1700;
	.TreasureOffers[7] = 1900;
	.TreasureOffers[8] = 2200;
	.TreasureOffers[9] = 2500;
	.TreasureOffers[10] = 4200;
	.TreasureOffers[11] = 4800;
	.TreasureOffers[12] = 5400;
	.TreasureOffers[13] = 6000;
	.TreasureOffers[14] = 6600;
	.TreasureOffers[15] = 10000;
	.TreasureOffers[16] = 22000;
	.TreasureOffers[17] = 42000;
	.TreasureOffers[18] = 45000;
	.TreasureOffers[19] = 70000;
	.TreasureOffers[20] = 112000;
	setunittitle(getnpcid(0), "<Treasure Collector>");
	end;
}

function	script	F_OfflineMvpCardAlbum	{
	.@size = bmc_mvpcount();
	if (.@size <= 0) {
		message strcharinfo(0), "The MVP Card Album is not ready yet.";
		getitem 35000,1;
		return;
	}
	getitem bmc_mvpid(rand(.@size)),1;
	return;
}

// Bodyguards: visible and proximity-reactive, but they do not open dialogue.
paramk,41,161,2	script	Guard#card_collector_1	405,2,1,{
	callsub S_Warn;
	end;

OnTouch:
	callsub S_Warn;
	end;

S_Warn:
	if (@CardGuardWarningUntil > gettimetick(2))
		return;
	@CardGuardWarningUntil = gettimetick(2) + 3;

	switch (rand(3)) {
	case 0:
		npctalk "Easy. Don't try anything.", strnpcinfo(0), bc_self;
		break;
	case 1:
		npctalk "Watch yourself. The Boss is protected.", strnpcinfo(0), bc_self;
		break;
	default:
		npctalk "Keep your hands where I can see them.", strnpcinfo(0), bc_self;
		break;
	}
	return;
}
