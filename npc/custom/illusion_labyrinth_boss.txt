//===== Ragnarok Offline Script =======================================
//= Illusion of Labyrinth - Chaotic Acolytes and Baphomet cycle
//====================================================================
// Acolytes always respawn after 5 seconds. Only boss summoning is locked
// for 2 hours after Baphomet dies; kills during that lock do not queue.

prt_mz03_i	monster	Chaotic Andrea	20521,1,5000,0,"prt_mz03_i_boss::OnAcolyteDead"
prt_mz03_i	monster	Chaotic Anes	20522,1,5000,0,"prt_mz03_i_boss::OnAcolyteDead"
prt_mz03_i	monster	Chaotic Silvano	20523,1,5000,0,"prt_mz03_i_boss::OnAcolyteDead"
prt_mz03_i	monster	Chaotic Cecilia	20524,1,5000,0,"prt_mz03_i_boss::OnAcolyteDead"

-	script	prt_mz03_i_boss	-1,{
	end;
OnAcolyteDead:
	if (playerattached() < 1)
		end;
	if (.boss_spawn || .cooldown)
		end;
	if (killedrid != 20521 && killedrid != 20522 && killedrid != 20523 && killedrid != 20524)
		end;
	if (.count[killedrid] == 1)
		end;
	.count[killedrid] = 1;
	if (.count[20521] != 1 || .count[20522] != 1 || .count[20523] != 1 || .count[20524] != 1)
		end;
	.count[20521] = 0;
	.count[20522] = 0;
	.count[20523] = 0;
	.count[20524] = 0;
	getunitdata killedgid, .@data;
	$@mobid[0] = 0;
	monster "prt_mz03_i", .@data[UMOB_X], .@data[UMOB_Y], "--ja--", 20520, 1, "prt_mz03_i_boss::OnBossDeath";
	if ($@mobid[0] < 1) {
		debugmes "prt_mz03_i_boss: ILL_BAPHOMET spawn failed";
		end;
	}
	.boss_gid = $@mobid[0];
	.boss_spawn = true;
	.invuln = false;
	mapannounce "prt_mz03_i", "A dreadful being has descended from beyond the chaos of the abyss.", bc_map, "0x70dbdb";
	stopnpctimer;
	initnpctimer;
	end;
OnBossDeath:
	if (.invuln && .boss_gid && unitexists(.boss_gid))
		setunitdata .boss_gid, UMOB_DMGIMMUNE, 0;
	.invuln = false;
	.boss_gid = 0;
	.boss_spawn = false;
	if (.cooldown)
		end;
	.cooldown = true;
	stopnpctimer;
	initnpctimer;
	end;
OnTimer45000:
	if (.cooldown || !.boss_spawn || !.boss_gid || !unitexists(.boss_gid))
		end;
	setunitdata .boss_gid, UMOB_DMGIMMUNE, 1;
	.invuln = true;
	mapannounce "prt_mz03_i", "Chaotic Baphomet becomes invincible.", bc_map, "0x70dbdb";
	end;
OnTimer55000:
	if (.cooldown)
		end;
	if (!.boss_spawn || !.boss_gid || !unitexists(.boss_gid)) {
		.invuln = false;
		.boss_spawn = false;
		.boss_gid = 0;
		end;
	}
	if (.invuln) {
		setunitdata .boss_gid, UMOB_DMGIMMUNE, 0;
		mapannounce "prt_mz03_i", "The invincibility from Chaotic Baphomet has been lifted.", bc_map, "0x70dbdb";
	}
	.invuln = false;
	initnpctimer;
	end;
OnTimer7200000:
	if (!.cooldown)
		end;
	.cooldown = false;
	stopnpctimer;
	end;
}
