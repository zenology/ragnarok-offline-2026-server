//===== Ragnarok Offline =====================================
//= Black-Market Consumable Seller
//===== Description: =========================================
//= Rooke sells a curated cash-origin consumable whitelist for
//= character-local Black Market Points. Each category has one hidden
//= pointshop per player-progression price tier, so simultaneous players
//= never share a catalog or price. A zero tier price means the SKU is not
//= sold in that tier. Full-row exclusions are absent from the catalog.
//============================================================

		mes "A kid?";
		mes "...No. You don't feel like a kid.";

	}
	mes "[Rooke]";

	mes "You look like you need something.";
	mes "I can get medicine. Almost every kind.";
	mes "The good kind. The kind the Kafra ladies pretend not to know about.";
	next;
	mes "[Rooke]";
	mes "Manuals, gum, tonics, tickets... stones, if you know how to ask.";
	mes "If it goes in a bottle, under the tongue, or into a forge, I can source it.";
	mes "First taste is honest. ^0055FFBlack Market Points^000000 only.";
	mes "Tell me what you're craving.";
	next;
	switch (select("Premium Service:EXP and Drop:Combat Buffs:Recovery:Convenience:Upgrade:Style and Reset:")) {
	case 1:
		callsub S_OpenCategory, 6;
		end;
	case 2:
		callsub S_OpenCategory, 0;
		end;
	case 3:
		callsub S_OpenCategory, 1;
		end;
	case 4:
		callsub S_OpenCategory, 2;
		end;
	case 5:
		callsub S_OpenCategory, 3;
		end;
	case 6:
		callsub S_OpenCategory, 5;
		end;
	case 7:
		callsub S_OpenCategory, 4;
		end;
	}
	end;

S_OpenCategory:
	.@cat = getarg(0);
	.@tier = callsub(S_GetTier);
	if (!getvariableofnpc(.CatHas[.@tier * 7 + .@cat], "cons_seller_db")) {
		clear;
		mes "[Rooke]";
		mes "Tch. Empty bottle. Ask me for something else.";
		close;
	}
	close2;
	callshop "cons" + .@cat + "#T" + .@tier, 1;
	end;

S_GetTier:
	if (BaseLevel >= 201)
		return 5;
	if (BaseLevel >= 100)
		return 4;
	.@rebirth = (eaclass() & EAJL_UPPER) != 0;
	if (BaseLevel <= 50)
		return .@rebirth ? 2 : 0;
	return .@rebirth ? 3 : 1;

OnInit:
	setunittitle(getnpcid(0), "<Alchemist>");
	end;
}
