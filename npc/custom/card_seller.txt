//===== rAthena Script =======================================
//= Monster Card Sellers A-Z
//===== Description: =========================================
//= Separate Normal, Miniboss, and MVP Card Sellers use character-local Black Market Points.
//= Stock and Collector data are loaded from black_market_catalog/card-*.yml.
//= Cards are grouped alphabetically. A 0-9 drawer appears before A
//= when a display name starts with a digit. Normal, Miniboss, and MVP cards
//= use their catalog CollectorOffer as the assessed base price without markup.
//= Daily Rate and character level/rebirth adjustments still apply.
//=
//= Uses a reviewed card-to-canonical-source mapping.
//===== Changelogs: ==========================================
//= 1.0 First version [AnnieRuru]
//= 1.1 Minor edits [Euphy]
//= 1.2 Update for monster mode and enchants [Lemongrass]
//= 2.0 Offline Black Market Point catalog; Normal cards only
//= 2.1 Level-based 10-50 Black Market Point pricing
//= 3.0 Add Miniboss and MVP Card Sellers
//= 3.1 Add curated special cards to the MVP Card Seller
//= 3.2 Fix MVP shop initialization
//= 3.3 Shop prices = Collector offer + 30%
//= 3.4 Special cards = midpoint MVP offer + 30%
//= 3.5 Veynar browse gates use Collector prices, not shop prices
//= 3.6 Opening dialogue no longer recites shop rates
//= 3.7 Digit-leading names use a 0-9 drawer before A
//= 3.8 Remove the 30% markup; all four catalogs use their assessed values
//============================================================

function	script	F_OfflineCardLerp	{
	.@lv = getarg(0);
	.@lv0 = getarg(1);
	.@lv1 = getarg(2);
	.@p0 = getarg(3);
	.@p1 = getarg(4);
	if (.@lv1 <= .@lv0)
		return .@p0;
	if (.@lv <= .@lv0)
		return .@p0;
	if (.@lv >= .@lv1)
		return .@p1;
	.@span = .@lv1 - .@lv0;
	return .@p0 + ((.@lv - .@lv0) * (.@p1 - .@p0) + .@span / 2) / .@span;
}

function	script	F_OfflineCardCollectorOffer	{
	.@lv = getarg(0);
	.@tier = getarg(1);
	if (.@lv < 1)
		return 0;
	switch (.@tier) {
	case 3:
		return callfunc("F_OfflineCardLerp", .@lv, 25, 255, 20000, 100000);
	case 2:
		return callfunc("F_OfflineCardLerp", .@lv, 27, 186, 7000, 20000);
	case 1:
		if (.@lv <= 50)
			return callfunc("F_OfflineCardLerp", .@lv, 1, 50, 250, 1000);
		if (.@lv <= 100)
			return callfunc("F_OfflineCardLerp", .@lv, 51, 100, 2000, 4000);
		return callfunc("F_OfflineCardLerp", .@lv, 101, 255, 7000, 15000);
	}
	return 0;
}

function	script	F_OfflineCardShopPrice	{
	.@offer = callfunc("F_OfflineCardCollectorOffer", getarg(0), getarg(1));
	if (.@offer < 1)
		return 0;
	return .@offer;
}

function	script	F_OfflineMvpMidCollectorOffer	{
	.@lo = callfunc("F_OfflineCardCollectorOffer", 25, 3);
	.@hi = callfunc("F_OfflineCardCollectorOffer", 255, 3);
	return (.@lo + .@hi) / 2;
}

function	script	F_OfflineSpecialCardShopPrice	{
	.@mid = callfunc("F_OfflineMvpMidCollectorOffer");
	if (.@mid < 1)
		return 0;
	return .@mid;
}

function	script	F_OfflineLetterShop	{
	// 0-25 = A-Z, 26 = leading digit (shop suffix "0"), -1 = skip.
	.@name$ = getarg(0);
	.@ch$ = strtoupper(charat(.@name$, 0));
	.@idx = strpos("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@ch$);
	if (.@idx >= 0)
		return .@idx;
	if (strpos("0123456789", .@ch$) >= 0)
		return 26;
	if (getarg(1, 0) == 1)
		return -1;
	return 0;
}

function	script	F_OfflineCardBrowseKey	{
	.@letter = getarg(0);
	.@count = getarg(1);
	if (.@count < 100)
		return "ALL";
	if (.@count < 200) {
		if (.@letter >= 0 && .@letter <= 7)
			return "AH";
		if (.@letter >= 8 && .@letter <= 15)
			return "IP";
		return "QZ9";
	}
	if (.@letter == 26)
		return "0";
	return charat("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@letter);
}

	mes "My brother handles the dangerous merchandise.";
	mes "The man at the end handles things I would rather not ask about.";

	end;

	end;

		mes "You are not nearly seasoned enough for what I sell.";

		mes "[Veynar]";
		mes "Let the world leave a few more scars upon you.";
		mes "Survive long enough to become someone worth remembering.";
		mes "Then you may return.";
		close;
	}

		mes "[Veynar]";
		mes "A cat.";
		mes "A talking cat with enough scars to stand before me.";
		mes "Unexpected. Not unwelcome.";
		next;
	} else if (eaclass()&EAJL_BABY) {
		mes "[Veynar]";
		mes "A child?";
		mes "...How deceptive.";
		mes "There is something profoundly dangerous behind that small frame.";
		mes "Interesting. You may proceed.";
		next;
	}

	if (BlackMarketPoints < .@min_look) {
		mes "[Veynar]";
		mes "No. I do not open my collection for visitors with such an unimpressive purse.";
		mes "Come back when your purse has learned some dignity.";
		next;
		mes "[Veynar]";
		mes "^777777..." + .@min_look + " should be enough to earn a look.";
		mes "Buying from me is another matter.^000000";
		close;
	}
	mes "[Veynar]";
	mes "Ah, another treasure hunter.";
	mes "You have found the black market's finest card dealer.";
	mes "You thought luck would bring you every card you wanted?";
	mes "If luck were enough, you would not be standing before me.";
	next;
	mes "[Veynar]";
	mes "Only I can obtain these rarities.";
	mes "Normal cards? Miniboss cards?";
	mes "Trinkets for beginners and collectors with modest ambitions.";
	mes "The merchants beside me will gladly entertain you with those.";
	mes "Do not confuse their stock with mine.";
	mes "Which collection have you come to see?";
	next;
	.@collection$ = "";
	if (.@mvp_count > 0)
		.@collection$ = "Show me your MVP cards.:";
	if (.@special_count > 0)
		.@collection$ += "I know you have something more valuable.:";
	if (.@mvp_count == 0)
		.@choice = 2;
	else if (.@special_count == 0)
		.@choice = 1;
	else
		.@choice = select(.@collection$);
	switch (.@choice) {
	case 1:
		.@shop$ = callfunc("F_BlackMarketBrowse", "card-mvp");
		if (.@shop$ == "") close;
		callfunc "F_BlackMarketDailyRemark", "Veynar", $BlackMarketRateCards;
		close2;
		callblackmarketshop .@shop$, 1, $BlackMarketRateCards;
		end;
	case 2:
		clear;
		mes "[Veynar]";
		mes "^777777...This one has taste.^000000";
		mes "So, you know I keep something more valuable.";
		.@special_look = 60000;
		if (BlackMarketPoints < .@special_look) {
			next;
			mes "[Veynar]";
			mes "Taste without the means to indulge it is merely appetite.";
			close;
		}
		next;
		mes "[Veynar]";
		mes "Very well. You may look upon the cards no other merchant can offer.";
		next;
		.@shop$ = callfunc("F_BlackMarketBrowse", "card-special");
		if (.@shop$ == "") close;
		callfunc "F_BlackMarketDailyRemark", "Veynar", $BlackMarketRateCards;
		close2;
		callblackmarketshop .@shop$, 1, $BlackMarketRateCards;
		end;
	}
	end;

OnInit:
	setunittitle(getnpcid(0), "<MVP Card Seller>");
	end;
}
