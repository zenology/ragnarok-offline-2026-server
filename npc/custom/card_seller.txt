function	script	F_OfflineCardLerp	{
	.@lv = getarg(0);
	.@lv0 = getarg(1);
	.@lv1 = getarg(2);
	.@p0 = getarg(3);
	.@p1 = getarg(4);
	if (.@lv1 <= .@lv0)
		return .@p0;
	if (.@lv <= .@lv0)
		return .@p0;
	if (.@lv >= .@lv1)
		return .@p1;
	.@span = .@lv1 - .@lv0;
	return .@p0 + ((.@lv - .@lv0) * (.@p1 - .@p0) + .@span / 2) / .@span;
}

function	script	F_OfflineCardCollectorOffer	{
	.@lv = getarg(0);
	.@tier = getarg(1);
	if (.@lv < 1)
		return 0;
	switch (.@tier) {
	case 3:
		return callfunc("F_OfflineCardLerp", .@lv, 25, 255, 20000, 100000);
	case 2:
		return callfunc("F_OfflineCardLerp", .@lv, 27, 186, 7000, 20000);
	case 1:
		if (.@lv <= 50)
			return callfunc("F_OfflineCardLerp", .@lv, 1, 50, 250, 1000);
		if (.@lv <= 100)
			return callfunc("F_OfflineCardLerp", .@lv, 51, 100, 2000, 4000);
		return callfunc("F_OfflineCardLerp", .@lv, 101, 255, 7000, 15000);
	}
	return 0;
}

function	script	F_OfflineCardShopPrice	{
	.@offer = callfunc("F_OfflineCardCollectorOffer", getarg(0), getarg(1));
	if (.@offer < 1)
		return 0;
	return (.@offer * 13 + 5) / 10;
}

function	script	F_OfflineMvpMidCollectorOffer	{
	.@lo = callfunc("F_OfflineCardCollectorOffer", 25, 3);
	.@hi = callfunc("F_OfflineCardCollectorOffer", 255, 3);
	return (.@lo + .@hi) / 2;
}

function	script	F_OfflineSpecialCardShopPrice	{
	.@mid = callfunc("F_OfflineMvpMidCollectorOffer");
	if (.@mid < 1)
		return 0;
	return (.@mid * 13 + 5) / 10;
}

function	script	F_OfflineLetterShop	{
	// 0-25 = A-Z, 26 = leading digit (shop suffix "0"), -1 = skip.
	.@name$ = getarg(0);
	.@ch$ = strtoupper(charat(.@name$, 0));
	.@idx = strpos("ABCDEFGHIJKLMNOPQRSTUVWXYZ", .@ch$);
	if (.@idx >= 0)
		return .@idx;
	if (strpos("0123456789", .@ch$) >= 0)
		return 26;
	if (getarg(1, 0) == 1)
		return -1;
	return 0;
}
