// Outdoor storage follows jor_mbase's over_kafra SignBoardList marker.
// The indoor blanket and dealer retain their client SignBoardList positions.
// Fees and supplies follow local town-service conventions, not EP21 quest logic.
jor_mbase,220,165,3	script	Storage Box#ep21	4_WOODBOX,{
	mes "[Storage Box]";
	mes "This box provides access to your personal storage. Access costs 40 Zeny, or one Free Ticket for Kafra Storage.";
	.@choice = prompt("Open Storage.:Cancel.");
	if (.@choice == 255)
		end;
	if (.@choice != 1)
		close;

	if (!callfunc("F_CanOpenStorage")) {
		next;
		mes "[Storage Box]";
		mes "You need Basic Skill Level 6 or the Doram Basic Skill to use storage.";
		close;
	}
	if (!countitem(7059) && Zeny < 40) {
		next;
		mes "[Storage Box]";
		mes "You need 40 Zeny or a Free Ticket for Kafra Storage.";
		close;
	}
	if (#kafra_code) {
		next;
		callfunc "F_CheckKafCode";
	}
	if (countitem(7059))
		delitem 7059,1;
	else {
		Zeny -= 40;
		RESRVPTS += 8;
	}
	close2;
	openstorage;
	end;
}
