//===== Ragnarok Offline =====================================
//= Black-Market Hat Seller
//===== Description: =========================================
//= Sells reviewed iRO cash/event headgear for Cash Points.
//= Browse Equipment, Rental, or Event by Top / Eye / Mouth, then 0-9
//= (when a display name starts with a digit)
//= then A-Z by client identifiedDisplayName. Letter shops are
//= filled in that display-name order, then ID.
//= Specialty opens its named equipment-family drawer directly.
//= Event is Racing Caps, RWC/RTC/RCC tournament hats, and iRO-history
//= create-pass leftovers. Display names do not keep [Event]/[S2]/[S3]
//= prefixes. Hats that exist
//= in item_db but have no client itemInfo.lua row stay on the
//= shelf; client itemInfo is a later pass. Hats with no worn accessory
//= model in the extracted trees are not stocked (2026-08-20). Client-tagged rental hats
//= are sold as duration boxes, not the loose SKU. Every SKU uses the
//= KB equipment-pricing taxonomy; Rental prices include its 30% discount.
//============================================================

paramk,41,173,2	script	Harlan#offline	897,{
	.@eac = eaclass();
	if ((.@eac&EAJ_BASEMASK) == EAJ_SUMMONER) {
		mes "[Harlan]";
		mes "A talking cat. In a milliner's stall.";
		mes "Sit still. I can still measure a head that talks.";
		next;
	} else if (.@eac&EAJL_BABY) {
		mes "[Harlan]";
		mes "You look like a child.";
		mes "You do not feel like one.";
		mes "I have sold hats to kings who frightened me less.";
		next;
	} else if (.@eac&EAJL_UPPER) {
		mes "[Harlan]";
		mes "That face is young. The way you stand is not.";
		mes "I remember customers who came back after the world remade them.";
		next;
	}
	mes "[Harlan]";
	mes "Do not brush past. Hats are not an afterthought.";
	mes "Crowns. Glasses. The little things people hide in their mouths.";
	mes "I sell the hat itself. Timed boxes, if you only need it for a while.";
	mes "Fitted circlets from harder work, if you would rather not fetch them.";
	mes "Tournament leftovers, if you missed the crowd.";
	mes "^0055FFCash Points^000000 only. No eggs.";
	next;
	switch (select("Equipment:Rental:Specialty:Event:")) {
	case 1:
		callsub S_OpenCatalog, 0;
		end;
	case 2:
		callsub S_OpenCatalog, 1;
		end;
	case 3:
		callsub S_OpenCatalog, 2;
		end;
	case 4:
		callsub S_OpenCatalog, 3;
		end;
	}
	end;

S_OpenCatalog:
	.@kind = getarg(0);
	// Specialty has only top headgear and groups by family, so do not make
	// players select a meaningless Top / Eye / Mouth menu first.
	.@slot = 0;
	if (.@kind != 2)
		.@slot = select("Top:Eye:Mouth:") - 1;
	if (.@kind == 0) {
		.@menu$ = getvariableofnpc(.HatMenu$[.@slot], "hat_seller_db");
		.@prefix$ = "hat";
	} else if (.@kind == 1) {
		.@menu$ = getvariableofnpc(.RentMenu$[.@slot], "hat_seller_db");
		.@prefix$ = "rent";
	} else if (.@kind == 2) {
		.@menu$ = getvariableofnpc(.SpecMenu$[.@slot], "hat_seller_db");
		.@prefix$ = "spec";
	} else {
		.@menu$ = getvariableofnpc(.EvtMenu$[.@slot], "hat_seller_db");
		.@prefix$ = "evt";
	}
	if (.@menu$ == "") {
		mes "[Harlan]";
		mes "That drawer is empty. Try another.";
		close;
	}
	.@s = select(.@menu$) - 1;
	.@idx = .@slot * 27 + .@s;
	if (.@kind == 0)
		.@letter$ = getvariableofnpc(.HatLetter$[.@idx], "hat_seller_db");
	else if (.@kind == 1)
		.@letter$ = getvariableofnpc(.RentLetter$[.@idx], "hat_seller_db");
	else if (.@kind == 2)
		.@letter$ = getvariableofnpc(.SpecLetter$[.@idx], "hat_seller_db");
	else
		.@letter$ = getvariableofnpc(.EvtLetter$[.@idx], "hat_seller_db");
	close2;
	callshop .@prefix$ + .@slot + "#" + .@letter$, 1;
	end;

OnInit:
	setunittitle(getnpcid(0), "<Milliner>");
	end;
}

		.@letter_index = strpos(.@alphabet$, .@letter$);
		for (.@slot = 0; .@slot < 3; ++.@slot) {
			if (.@loc & .@eqp[.@slot]) {
				if (.@letter_index == -2)
					.SpecHasNum[.@slot] = true;
				else
					.SpecHas[.@slot * 26 + .@letter_index] = true;
				npcshopadditem "spec" + .@slot + "#" + .@letter$, .@id, .@price;
			}
		}
	}

	.@kept = 0;
	for (.@i = 0; .@i < .EvtCount; ++.@i) {
		.@id = .EvtId[.@i];
		.@name$ = getitemname(.@id);
		// Empty / "Unknown Item" / "null" means the ID is missing from item_db. This checkout's getitemname returns "null" for a missing ID, not "Unknown Item". That is not a missing client itemInfo.lua row.
		if (.@name$ == "" || .@name$ == "Unknown Item" || .@name$ == "null") {
			debugmes "Hat Seller: missing event item " + .@id + ".";
			continue;
		}
		if (getiteminfo(.@id, ITEMINFO_TYPE) != IT_ARMOR) {
			debugmes "Hat Seller: rejected type for event item " + .@id + " (" + .@name$ + ").";
			continue;
		}
		.@loc = getiteminfo(.@id, ITEMINFO_LOCATIONS);
		if (.@loc & .@costume) {
			debugmes "Hat Seller: rejected costume slot for event item " + .@id + " (" + .@name$ + ").";
			continue;
		}
		.EvtId[.@kept] = .@id;
		.EvtPrice[.@kept] = .EvtPrice[.@i];
		.EvtAegis$[.@kept] = getiteminfo(.@id, ITEMINFO_AEGISNAME);
		.@disp$ = .EvtDisp$[.@i];
		if (.@disp$ == "")
			.@disp$ = .@name$;
		.EvtDisp$[.@kept] = .@disp$;
		.@evtname$[.@kept] = strtolower(.@disp$);
		.@kept += 1;
	}
	.EvtCount = .@kept;
	for (.@i = 0; .@i < .EvtCount; ++.@i) {
		for (.@j = .@i + 1; .@j < .EvtCount; ++.@j) {
			.@cmp = strcmp(.@evtname$[.@j], .@evtname$[.@i]);
			if (.@cmp < 0 || (.@cmp == 0 && .EvtId[.@j] < .EvtId[.@i])) {
				.@tmp = .EvtId[.@i];
				.EvtId[.@i] = .EvtId[.@j];
				.EvtId[.@j] = .@tmp;
				.@tmp = .EvtPrice[.@i];
				.EvtPrice[.@i] = .EvtPrice[.@j];
				.EvtPrice[.@j] = .@tmp;
				.@tmp$ = .EvtAegis$[.@i];
				.EvtAegis$[.@i] = .EvtAegis$[.@j];
				.EvtAegis$[.@j] = .@tmp$;
				.@tmp$ = .EvtDisp$[.@i];
				.EvtDisp$[.@i] = .EvtDisp$[.@j];
				.EvtDisp$[.@j] = .@tmp$;
				.@tmp$ = .@evtname$[.@i];
				.@evtname$[.@i] = .@evtname$[.@j];
				.@evtname$[.@j] = .@tmp$;
			}
		}
	}
	for (.@i = 0; .@i < .EvtCount; ++.@i) {
		.@id = .EvtId[.@i];
		.@price = .EvtPrice[.@i];
		.@name$ = .EvtDisp$[.@i];
		.@loc = getiteminfo(.@id, ITEMINFO_LOCATIONS);
		.@letter$ = strtoupper(charat(.@name$, 0));
		.@letter_index = strpos(.@alphabet$, .@letter$);
		if (.@letter_index < 0) {
			if (strpos("0123456789", .@letter$) >= 0) {
				.@letter$ = "0";
				.@letter_index = -2;
			} else {
				.@letter$ = "A";
				.@letter_index = 0;
			}
		}
		for (.@slot = 0; .@slot < 3; ++.@slot) {
			if (.@loc & .@eqp[.@slot]) {
				if (.@letter_index == -2)
					.EvtHasNum[.@slot] = true;
				else
					.EvtHas[.@slot * 26 + .@letter_index] = true;
				npcshopadditem "evt" + .@slot + "#" + .@letter$, .@id, .@price;
			}
		}
	}


	for (.@slot = 0; .@slot < 3; ++.@slot) {
		.HatMenu$[.@slot] = "";
		.RentMenu$[.@slot] = "";
		.SpecMenu$[.@slot] = "";
		.EvtMenu$[.@slot] = "";
		.HatLetterCount[.@slot] = 0;
		.RentLetterCount[.@slot] = 0;
		.SpecLetterCount[.@slot] = 0;
		.EvtLetterCount[.@slot] = 0;
		if (.HatHasNum[.@slot]) {
			.HatLetter$[.@slot * 27] = "0";
			.HatMenu$[.@slot] += "0-9:";
			.HatLetterCount[.@slot] = 1;
		}
		if (.RentHasNum[.@slot]) {
			.RentLetter$[.@slot * 27] = "0";
			.RentMenu$[.@slot] += "0-9:";
			.RentLetterCount[.@slot] = 1;
		}
		for (.@family = 0; .@family < .SpecFamilyCount; ++.@family) {
			.@letter$ = .SpecFamilyCode$[.@family];
			.@letter_index = strpos(.@alphabet$, .@letter$);
			if (.SpecHas[.@slot * 26 + .@letter_index]) {
				.SpecLetter$[.@slot * 27 + .SpecLetterCount[.@slot]] = .@letter$;
				.SpecMenu$[.@slot] += .SpecFamilyName$[.@family] + ":";
				.SpecLetterCount[.@slot] += 1;
			}
		}
		if (.EvtHasNum[.@slot]) {
			.EvtLetter$[.@slot * 27] = "0";
			.EvtMenu$[.@slot] += "0-9:";
			.EvtLetterCount[.@slot] = 1;
		}
		for (.@i = 0; .@i < 26; ++.@i) {
			if (.HatHas[.@slot * 26 + .@i]) {
				.@letter$ = charat(.@alphabet$, .@i);
				.@n = .HatLetterCount[.@slot];
				.HatLetter$[.@slot * 27 + .@n] = .@letter$;
				.HatMenu$[.@slot] += .@letter$ + ":";
				.HatLetterCount[.@slot] = .@n + 1;
			}
			if (.RentHas[.@slot * 26 + .@i]) {
				.@letter$ = charat(.@alphabet$, .@i);
				.@n = .RentLetterCount[.@slot];
				.RentLetter$[.@slot * 27 + .@n] = .@letter$;
				.RentMenu$[.@slot] += .@letter$ + ":";
				.RentLetterCount[.@slot] = .@n + 1;
			}
			if (.EvtHas[.@slot * 26 + .@i]) {
				.@letter$ = charat(.@alphabet$, .@i);
				.@n = .EvtLetterCount[.@slot];
				.EvtLetter$[.@slot * 27 + .@n] = .@letter$;
				.EvtMenu$[.@slot] += .@letter$ + ":";
				.EvtLetterCount[.@slot] = .@n + 1;
			}
		}
	}
	freeloop 0;
	end;
}
