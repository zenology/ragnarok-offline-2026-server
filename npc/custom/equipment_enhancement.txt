// Vesper replaces every native Random Option on one equipped ordinary item.
// Pool data is generated into equipment_enhancement_db.txt.

paramk,43,181,4	script	Vesper#bm_enhance	4_F_BERRYTEA,{
	disable_items;
	mes "[Vesper]";
	mes "Excellent... a properly equipped test subject.";
	mes "I shall replace every Random Option on one ordinary equipped item.";
	mes "Do not panic: your item will not break. A reroll merely clears all five current option slots.";
	next;

	.@count = 0;
	for (.@scan = 0; .@scan < 10; ++.@scan) {
		.@slot = (.@scan == 0 ? EQI_HEAD_TOP : .@scan == 1 ? EQI_HEAD_MID : .@scan == 2 ? EQI_HEAD_LOW : .@scan == 3 ? EQI_HAND_R : .@scan == 4 ? EQI_HAND_L : .@scan == 5 ? EQI_ARMOR : .@scan == 6 ? EQI_GARMENT : .@scan == 7 ? EQI_SHOES : .@scan == 8 ? EQI_ACC_L : EQI_ACC_R);
		.@item = getequipid(.@slot);
		if (.@item < 0) continue;
		.@type = getiteminfo(.@item, ITEMINFO_TYPE);
		if (.@type != IT_WEAPON && .@type != IT_ARMOR) continue;
		if (.@type == IT_WEAPON && .@slot != EQI_HAND_R && .@slot != EQI_HAND_L) continue;
		if (.@type == IT_ARMOR && .@slot == EQI_HAND_R) continue;
		.@uid$ = getequipuniqueid(.@slot);
		if (.@uid$ == "") continue;
		.@label$ = (.@slot == EQI_HEAD_TOP ? "Top" : .@slot == EQI_HEAD_MID ? "Middle" : .@slot == EQI_HEAD_LOW ? "Lower" : .@slot == EQI_HAND_R ? "Weapon Right" : .@slot == EQI_HAND_L && .@type == IT_WEAPON ? "Weapon Left" : .@slot == EQI_HAND_L ? "Shield" : .@slot == EQI_ARMOR ? "Armor" : .@slot == EQI_GARMENT ? "Garment" : .@slot == EQI_SHOES ? "Shoes" : .@slot == EQI_ACC_L ? "Accessory Left" : "Accessory Right");
		.@found = -1;
		for (.@i = 0; .@i < .@count; ++.@i)
			if (.@chosen_uid$[.@i] == .@uid$) .@found = .@i;
		if (.@found >= 0) {
			.@chosen_label$[.@found] += "/" + .@label$;
			continue;
		}
		.@chosen_slot[.@count] = .@slot;
		.@chosen_type[.@count] = .@type;
		.@chosen_uid$[.@count] = .@uid$;
		.@chosen_label$[.@count] = .@label$;
		.@chosen_item[.@count] = .@item;
		++.@count;
	}
	if (.@count < 1) {
		mes "Wear an ordinary weapon, shield, armor, headgear, garment, shoes, or accessory first.";
		mes "Costumes, Shadow Equipment, and ammunition cannot be enhanced here.";
		close;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < .@count; ++.@i)
		.@menu$ += .@chosen_label$[.@i] + " - " + getitemname(.@chosen_item[.@i]) + ":";
	.@choice = select(.@menu$ + "Cancel") - 1;
	if (.@choice == .@count) close;
	clear;
	.@slot = .@chosen_slot[.@choice];
	.@type = .@chosen_type[.@choice];
	.@uid$ = .@chosen_uid$[.@choice];
	.@item = .@chosen_item[.@choice];
	.@existing = 0;
	for (.@i = 0; .@i < 5; ++.@i)
		if (getequiprandomoption(.@slot, .@i, ROA_ID) != 0) ++.@existing;
	.@price = blackmarketprice(.@existing > 0 ? 2000 : 1000);
	if (.@price <= 0) {
		mes "[Vesper]";
		mes "The ledger refuses to price this experiment safely. No points were charged.";
		close;
	}

	mes "[Vesper]";
	mes "Ah, ^0055FF" + getitemname(.@item) + "^000000 (" + .@chosen_label$[.@choice] + "). A splendid specimen.";
	mes "Current Random Options: " + .@existing + ". Price: ^0055FF" + .@price + " Black Market Points^000000.";
	mes "This experiment replaces all five option slots. Every existing Random Option will be cleared.";
	if (select("Replace all options:Cancel") == 2) close;
	clear;

	.@rolled = 0;
	if (rand(100) < 70) {
		.@rolled = 1;
		if (rand(100) < 60) {
			.@rolled = 2;
			if (rand(100) < 50) {
				.@rolled = 3;
				if (rand(100) < 20) {
					.@rolled = 4;
					if (rand(100) < 10) .@rolled = 5;
				}
			}
		}
	}
	for (.@i = 0; .@i < 5; ++.@i) {
		.@roll_id[.@i] = 0;
		.@roll_value[.@i] = 0;
		.@roll_param[.@i] = 0;
	}
	.@pool_offset = .@type == IT_WEAPON ? 0 : 5;
	for (.@slot_index = 0; .@slot_index < .@rolled; ++.@slot_index) {
		.@pool = .@pool_offset + .@slot_index;
		.@start = getelementofarray(getvariableofnpc(.PoolStart[0], "equipment_enhancement_db"), .@pool);
		.@size = getelementofarray(getvariableofnpc(.PoolCount[0], "equipment_enhancement_db"), .@pool);
		.@candidate_count = 0;
		for (.@i = .@start; .@i < .@start + .@size; ++.@i) {
			.@candidate_id = getelementofarray(getvariableofnpc(.OptionId[0], "equipment_enhancement_db"), .@i);
			if (inarray(.@roll_id[0], .@candidate_id) == -1)
				.@candidate_index[.@candidate_count++] = .@i;
		}
		if (.@candidate_count < 1) {
			mes "[Vesper]";
			mes "How maddening! The option registry is invalid. No points were charged and your item was not changed.";
			close;
		}
		if (.@slot_index == 0 && inarray(getvariableofnpc(.CrimsonItemId[0], "equipment_enhancement_db"), .@item) != -1) {
			// Registry generator enforces the ten approved Crimson elements.
			.@element_count = 10;
			.@weight_total = 0;
			for (.@i = 0; .@i < .@element_count; ++.@i)
				.@weight_total += getelementofarray(getvariableofnpc(.CrimsonOptionWeight[0], "equipment_enhancement_db"), .@i);
			.@element_roll = rand(.@weight_total);
			for (.@i = 0; .@i < .@element_count; ++.@i) {
				.@element_roll -= getelementofarray(getvariableofnpc(.CrimsonOptionWeight[0], "equipment_enhancement_db"), .@i);
				if (.@element_roll < 0) break;
			}
			.@roll_id[.@slot_index] = getelementofarray(getvariableofnpc(.CrimsonOptionId[0], "equipment_enhancement_db"), .@i);
			.@roll_value[.@slot_index] = 0;
		} else {
			.@selected_index = .@candidate_index[rand(.@candidate_count)];
			.@roll_id[.@slot_index] = getelementofarray(getvariableofnpc(.OptionId[0], "equipment_enhancement_db"), .@selected_index);
			.@minimum = getelementofarray(getvariableofnpc(.OptionMin[0], "equipment_enhancement_db"), .@selected_index);
			.@maximum = getelementofarray(getvariableofnpc(.OptionMax[0], "equipment_enhancement_db"), .@selected_index);
			.@roll_value[.@slot_index] = .@minimum + rand(.@maximum - .@minimum + 1);
		}
	}

	.@result = blackmarketenhance(.@slot, .@uid$, .@roll_id, .@roll_value, .@roll_param);
	mes "[Vesper]";
	if (.@result == 1) {
		if (.@rolled == 0)
			mes "Fascinating! The replacement yielded zero options. Your item is intact; a later attempt uses the no-options price.";
		else
			mes "Magnificent! Your item now has " + .@rolled + " Random Option" + (.@rolled == 1 ? "." : "s.");
	} else if (.@result == 2)
		mes "The specimen changed before I could begin. No Black Market Points were charged.";
	else if (.@result == 3)
		mes "Insufficient Black Market Points. Science has standards; no changes were made.";
	else
		mes "The generated option data was invalid. No Black Market Points were charged and your item was not changed.";
	close;
}
