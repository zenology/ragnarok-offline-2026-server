paramk,37,163,2	script	Mordain#offline	4_EP21_GALAXIA_A,{
	if (!bmc_ready("mordain")) {
		mes "[Mordain]";
		mes "This stall's catalog is unavailable. Please try again later.";
		close;
	}
	mes "[Mordain]";
	mes "Body armor, sorted for the discerning knight.";
	.@shop$ = callfunc("F_BlackMarketBrowse", "mordain");
	if (.@shop$ == "") { close; }
	callfunc "F_BlackMarketDailyRemark", "Mordain", $BlackMarketRateMordain;
	close2;
	callblackmarketshop .@shop$, 1, $BlackMarketRateMordain;
	end;

OnInit:
	setunittitle(getnpcid(0), "<Dark Knight>");
	end;
}
