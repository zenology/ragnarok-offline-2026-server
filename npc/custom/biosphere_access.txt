// Native main-story completion is stage 33; later stages and EP17 tickets pass.
function	script	F_BioRequireEp172	{
	if (ep17_2_main >= 33)
		return;
	mes "[Biosphere Access]";
	mes "Complete the Episode 17.2 main story before entering Varmundt's Biosphere.";
	close;
}

}

// Each original blue gate has its own Depth 1 landing point (28274-28286).
function	script	F_BioDepthEntry	{
	callfunc "F_BioRequireEp172";
	if (!biosphere_depth_intro) {
		mes "[Depth Entrance]";
		mes "Speak with the Research Manager inside the mansion before entering the Biosphere depths.";
		close;
	}
	mes "[Depth Entrance]";
	mes "This gate leads to Biosphere Depth 1. Are you sure you want to enter?";
	next;
	if (select("Enter:Not yet.") == 2)
		close;
	close2;
	warp "bl_depth1",getarg(0),getarg(1);
	return;
}

// Four selectable landings confirmed by the original Abyss update and NAVI.
ba_chess,13,36,0	script	Abyss Entrance#biosphere	GATE_SKYBLUE,{
	callfunc "F_BioRequireEp172";
	if (!biosphere_depth_intro) {
		mes "[Abyss Entrance]";
		mes "Speak with the Research Manager inside the mansion before entering the Biosphere depths.";
		close;
	}
	mes "[Abyss Entrance]";
	mes "Enter the Biosphere Abyss? Choose an entrance direction.";
	next;
	.@entrance = select("2 o'clock:4 o'clock:8 o'clock:10 o'clock:Not yet.");
	if (.@entrance == 5)
		close;
	close2;
	switch (.@entrance) {
	case 1: warp "bl_depth2",194,188; break;
	case 2: warp "bl_depth2",190,64; break;
	case 3: warp "bl_depth2",57,87; break;
	case 4: warp "bl_depth2",90,202; break;
	}
	end;
}
