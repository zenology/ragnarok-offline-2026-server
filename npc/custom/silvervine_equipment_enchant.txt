function	script	F_SVE_DbInt	{
	.@idx = getarg(1);
	switch (getarg(0)) {
	case 0: return getelementofarray(getvariableofnpc(.ItemId[0], "silvervine_equipment_enchant_db"), .@idx);
	case 1: return getelementofarray(getvariableofnpc(.GroupId[0], "silvervine_equipment_enchant_db"), .@idx);
	case 2: return getelementofarray(getvariableofnpc(.MinRefine[0], "silvervine_equipment_enchant_db"), .@idx);
	case 3: return getelementofarray(getvariableofnpc(.MinEnchantgrade[0], "silvervine_equipment_enchant_db"), .@idx);
	case 4: return getelementofarray(getvariableofnpc(.AllowRandomOptions[0], "silvervine_equipment_enchant_db"), .@idx);
	case 5: return getelementofarray(getvariableofnpc(.GroupOrderStart[0], "silvervine_equipment_enchant_db"), .@idx);
	case 6: return getelementofarray(getvariableofnpc(.GroupOrderCount[0], "silvervine_equipment_enchant_db"), .@idx);
	case 7: return getelementofarray(getvariableofnpc(.GroupOrderCardIndex[0], "silvervine_equipment_enchant_db"), .@idx);
	case 8: return getelementofarray(getvariableofnpc(.PoolGroupIdx[0], "silvervine_equipment_enchant_db"), .@idx);
	case 9: return getelementofarray(getvariableofnpc(.PoolCardSlot[0], "silvervine_equipment_enchant_db"), .@idx);
	case 10: return getelementofarray(getvariableofnpc(.PoolEnchantgrade[0], "silvervine_equipment_enchant_db"), .@idx);
	case 11: return getelementofarray(getvariableofnpc(.PoolStart[0], "silvervine_equipment_enchant_db"), .@idx);
	case 12: return getelementofarray(getvariableofnpc(.PoolCount[0], "silvervine_equipment_enchant_db"), .@idx);
	case 13: return getelementofarray(getvariableofnpc(.PoolSlotChance[0], "silvervine_equipment_enchant_db"), .@idx);
	case 14: return getelementofarray(getvariableofnpc(.PoolItemId[0], "silvervine_equipment_enchant_db"), .@idx);
	case 15: return getelementofarray(getvariableofnpc(.PoolItemChance[0], "silvervine_equipment_enchant_db"), .@idx);
	}
	return 0;
}

function	script	F_SVE_FindRegistryIndex	{
	.@item_id = getarg(0);
	.@size = getvariableofnpc(.ItemCount, "silvervine_equipment_enchant_db");
	if (.@size < 1)
		return -1;
	for (.@i = 0; .@i < .@size; ++.@i) {
		if (getelementofarray(getvariableofnpc(.ItemId[0], "silvervine_equipment_enchant_db"), .@i) == .@item_id)
			return .@i;
	}
	return -1;
}

function	script	F_SVE_FindRoIndex	{
	.@item_id = getarg(0);
	.@size = getvariableofnpc(.RoItemCount, "silvervine_equipment_enchant_db");
	if (.@size < 1)
		return -1;
	for (.@i = 0; .@i < .@size; ++.@i) {
		if (getelementofarray(getvariableofnpc(.RoItemId[0], "silvervine_equipment_enchant_db"), .@i) == .@item_id)
			return .@i;
	}
	return -1;
}

function	script	F_SVE_GroupEnabled	{
	.@group_id = getarg(0);
	.@n = getvariableofnpc(.EnabledGroupCount, "silvervine_equipment_enchant_db");
	for (.@i = 0; .@i < .@n; ++.@i) {
		if (getelementofarray(getvariableofnpc(.EnabledGroupId[0], "silvervine_equipment_enchant_db"), .@i) == .@group_id)
			return 1;
	}
	return 0;
}

function	script	F_SVE_FindPoolIndex	{
	.@gidx = getarg(0);
	.@card_slot = getarg(1);
	.@grade = getarg(2);
	.@size = getvariableofnpc(.PoolRowCount, "silvervine_equipment_enchant_db");
	for (.@i = 0; .@i < .@size; ++.@i) {
		if (callfunc("F_SVE_DbInt", 8, .@i) != .@gidx)
			continue;
		if (callfunc("F_SVE_DbInt", 9, .@i) != .@card_slot)
			continue;
		if (callfunc("F_SVE_DbInt", 10, .@i) != .@grade)
			continue;
		return .@i;
	}
	return -1;
}

function	script	F_SVE_RollEnchant	{
	.@pool = getarg(0);
	if (.@pool < 0)
		return 0;
	.@start = callfunc("F_SVE_DbInt", 11, .@pool);
	.@count = callfunc("F_SVE_DbInt", 12, .@pool);
	if (.@count < 1)
		return 0;
	.@slot_chance = callfunc("F_SVE_DbInt", 13, .@pool);
	if (.@slot_chance < 100000 && rand(100000) > .@slot_chance)
		return -1;
	.@maximum = 3 * .@count;
	for (.@try = 0; .@try < .@maximum; ++.@try) {
		.@row = .@start + rand(.@count);
		.@chance = callfunc("F_SVE_DbInt", 15, .@row);
		if (rand(10000) < .@chance)
			return callfunc("F_SVE_DbInt", 14, .@row);
	}
	.@row = .@start + rand(.@count);
	return callfunc("F_SVE_DbInt", 14, .@row);
}

function	script	F_SVE_ChoosePoolEnchant	{
	.@pool = getarg(0);
	.@start = callfunc("F_SVE_DbInt", 11, .@pool);
	.@count = callfunc("F_SVE_DbInt", 12, .@pool);
	if (.@count < 1)
		return 0;
	freeloop(1);
	.@n = 0;
	for (.@r = 0; .@r < .@count; ++.@r) {
		.@id = callfunc("F_SVE_DbInt", 14, .@start + .@r);
		if (.@id < 1)
			continue;
		if (inarray(.@list[0], .@id) != -1)
			continue;
		.@list[.@n] = .@id;
		++.@n;
	}
	freeloop(0);
	if (.@n < 1)
		return 0;
	.@page_count = (.@n + 19) / 20;
	for (.@page = 0; .@page < .@page_count; ++.@page) {
		.@page_start = .@page * 20;
		.@menu$ = "";
		.@shown = 0;
		for (.@j = .@page_start; .@j < .@n && .@shown < 20; ++.@j) {
			.@menu$ += getitemname(.@list[.@j]) + ":";
			++.@shown;
		}
		if (.@page_start + .@shown < .@n)
			.@menu$ += "Next page:Cancel";
		else
			.@menu$ += "Cancel";
		.@choice = select(.@menu$) - 1;
		if (.@choice < 0 || .@choice > .@shown)
			return 0;
		if (.@choice == .@shown) {
			if (.@page + 1 < .@page_count)
				continue;
			return 0;
		}
		return .@list[.@page_start + .@choice];
	}
	return 0;
}

function	script	F_SVE_StepState	{
	// getarg(0)=equip_slot, getarg(1)=group_id, getarg(2)=ui_step (1-based)
	// Returns: 0=hidden, 1=new, 2=reroll
	.@equip_slot = getarg(0);
	.@group_id = getarg(1);
	.@ui_step = getarg(2);
	.@gidx = .@group_id - 1;
	.@order_start = callfunc("F_SVE_DbInt", 5, .@gidx);
	.@order_count = callfunc("F_SVE_DbInt", 6, .@gidx);
	if (.@ui_step < 1 || .@ui_step > .@order_count)
		return 0;
	for (.@p = 1; .@p < .@ui_step; ++.@p) {
		.@prev_card = callfunc("F_SVE_DbInt", 7, .@order_start + .@p - 1);
		.@prev_id = getequipcardid(.@equip_slot, .@prev_card);
		if (!F_IsCharm(.@prev_id))
			return 0;
	}
	.@card_idx = callfunc("F_SVE_DbInt", 7, .@order_start + .@ui_step - 1);
	.@item_id = getequipid(.@equip_slot);
	if (.@card_idx < getiteminfo(.@item_id, ITEMINFO_SLOT))
		return 0;
	.@card_id = getequipcardid(.@equip_slot, .@card_idx);
	if (.@card_id == 0)
		return 1;
	if (F_IsCharm(.@card_id))
		return 2;
	return 0;
}

function	script	F_SVE_HasActionableStep	{
	.@equip_slot = getarg(0);
	.@group_id = getarg(1);
	.@gidx = .@group_id - 1;
	.@order_count = callfunc("F_SVE_DbInt", 6, .@gidx);
	for (.@step = 1; .@step <= .@order_count; ++.@step) {
		if (callfunc("F_SVE_StepState", .@equip_slot, .@group_id, .@step))
			return 1;
	}
	return 0;
}

	if (callfunc("F_IsEquipIDHack", .@slot, .@eq_id) || callfunc("F_IsEquipCardHack", .@slot, .@card[0], .@card[1], .@card[2], .@card[3]) || callfunc("F_IsEquipRefineHack", .@slot, .@refine))
		close;

	if (.@pick) {
		.@enchant_id = callfunc("F_SVE_ChoosePoolEnchant", .@pool);
		if (.@enchant_id == 0)
			close;
	} else {
		.@enchant_id = callfunc("F_SVE_RollEnchant", .@pool);
		if (.@enchant_id < 0) {
			mes "[Silvervine Enchanter]";
			mes "Fizzled. Bad luck—happens to the best of us.";
			next;
			mes "[Silvervine Enchanter]";
			mes "No fruit taken. Try again if you've got the nerve.";
			close;
		}
		if (.@enchant_id == 0) {
			mes "[Silvervine Enchanter]";
			mes "Something's wrong in my book. Invalid charm.";
			next;
			mes "[Silvervine Enchanter]";
			mes "Nothing changed. Don't ask me to explain bureaucracy.";
			close;
		}
	}

	delitem 6417, .@pay;
	delequip .@slot;
	.@card[.@card_idx] = .@enchant_id;
	getitem4 .@eq_id, 1, 1, .@refine, 0, .@card[0], .@card[1], .@card[2], .@card[3], .@grade, .@OptID, .@OptVal, .@OptParam;

	mes "[Silvervine Enchanter]";
	mes "Done. ^0055FF" + getitemname(.@enchant_id) + "^000000 — slotted where you wanted.";
	next;
	mes "[Silvervine Enchanter]";
	mes "Treat it better than the last owner did, kid.";
	specialeffect2 EF_REPAIRWEAPON;
	close;
}
