// Malangdo Silvervine enchant (card-slot + Random Options). Pools: silvervine_enchant_item_pools.yml

function	script	F_SVE_ChoosePoolEnchant	{
	.@item_id = getarg(0);
	.@card_idx = getarg(1);
	.@count = sve_poolcount(.@item_id, .@card_idx);
	if (.@count < 1)
		return 0;
	freeloop(1);
	.@n = 0;
	for (.@r = 0; .@r < .@count; ++.@r) {
		.@id = sve_poolitem(.@item_id, .@card_idx, .@r);
		if (.@id < 1)
			continue;
		if (inarray(.@list[0], .@id) != -1)
			continue;
		.@list[.@n] = .@id;
		++.@n;
	}
	freeloop(0);
	if (.@n < 1)
		return 0;
	.@page_count = (.@n + 19) / 20;
	for (.@page = 0; .@page < .@page_count; ++.@page) {
		.@page_start = .@page * 20;
		.@menu$ = "";
		.@shown = 0;
		for (.@j = .@page_start; .@j < .@n && .@shown < 20; ++.@j) {
			.@menu$ += replacestr(getitemname(.@list[.@j]), ":", "") + ":";
			++.@shown;
		}
		if (.@page_start + .@shown < .@n)
			.@menu$ += "Next page:Cancel";
		else
			.@menu$ += "Cancel";
		.@choice = select(.@menu$) - 1;
		if (.@choice < 0 || .@choice > .@shown)
			return 0;
		if (.@choice == .@shown) {
			if (.@page + 1 < .@page_count)
				continue;
			return 0;
		}
		return .@list[.@page_start + .@choice];
	}
	return 0;
}

function	script	F_SVE_StepState	{
	// getarg(0)=equip_slot, getarg(1)=item_id, getarg(2)=ui_step (1-based)
	// Returns: 0=hidden, 1=new, 2=reroll
	.@equip_slot = getarg(0);
	.@item_id = getarg(1);
	.@ui_step = getarg(2);
	.@order_count = sve_stepcount(.@item_id);
	if (.@ui_step < 1 || .@ui_step > .@order_count)
		return 0;
	for (.@p = 1; .@p < .@ui_step; ++.@p) {
		.@prev_card = sve_stepcard(.@item_id, .@p);
		if (.@prev_card < 0)
			return 0;
		.@prev_id = getequipcardid(.@equip_slot, .@prev_card);
		if (!F_IsCharm(.@prev_id))
			return 0;
	}
	.@card_idx = sve_stepcard(.@item_id, .@ui_step);
	if (.@card_idx < 0)
		return 0;
	if (.@card_idx < getiteminfo(getequipid(.@equip_slot), ITEMINFO_SLOT))
		return 0;
	.@card_id = getequipcardid(.@equip_slot, .@card_idx);
	if (.@card_id == 0)
		return 1;
	if (F_IsCharm(.@card_id))
		return 2;
	return 0;
}

function	script	F_SVE_HasActionableStep	{
	.@equip_slot = getarg(0);
	.@item_id = getarg(1);
	.@order_count = sve_stepcount(.@item_id);
	for (.@step = 1; .@step <= .@order_count; ++.@step) {
		if (callfunc("F_SVE_StepState", .@equip_slot, .@item_id, .@step))
			return 1;
	}
	return 0;
}
