//===== Ragnarok Offline =====================================
//= Treasure Eater
//= Iron, Silver, and Gold treasure boxes contain coin-bug snacks.
//= Each exchange grants exclusively Silvervine or Cookie by box group.
//============================================================

malangdo,138,140,4	script	Moth#offline	4_CAT_ADV2,{
L_Main:
	clear;
	if (!.Valid) {
		mes "[Moth]";
		mes "My menu is a mess. Come back after someone fixes my dinner list.";
		close;
	}
	mes "[Moth]";
	mes "Mrrr... treasure smells delicious. Not the chest. The little coin bugs inside.";
	next;
	mes "[Moth]";
	mes "Sometimes I bite a real coin by mistake. Ow. My teeth are still complaining.";
	next;
	switch (select("Feed one treasure type.:Feed every accepted treasure.:Which treasures do you eat?:Why do you eat treasure?:Leave.")) {
	case 1: goto L_Feed;
	case 2: goto L_Bulk;
	case 3: goto L_Catalog;
	case 4: goto L_WhyTreasure;
	default: close;
	}

L_WhyTreasure:
	clear;
	mes "[Moth]";
	mes "Those shiny coins have legs. They are coin bugs pretending to be treasure.";
	next;
	mes "[Moth]";
	mes "I eat the bugs inside. The chest is just their lunch box.";
	next;
	mes "[Moth]";
	mes "Well... sometimes the coin does not have legs. I notice that after I bite it.";
	next;
	mes "[Moth]";
	mes "Real coins taste awful. And they hurt my teeth. Very rude of them to look like dinner.";
	next;
	mes "[Moth]";
	mes "Bring me Iron, Silver, or Gold Treasure Boxes. No cards, Diamond, or Platinum.";
	next;
	goto L_Main;

L_Bulk:
	clear;
	getinventorylist;
	deletearray .@bulk_ids[0], getarraysize(.@bulk_ids);
	deletearray .@bulk_amounts[0], getarraysize(.@bulk_amounts);
	deletearray .@bulk_groups[0], getarraysize(.@bulk_groups);
	setarray .@group_counts[0], 0,0,0,0,0;
	.@bulk_count = 0;
	.@bulk_treasures = 0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		.@item_id = @inventorylist_id[.@i];
		if (inarray(.@bulk_ids[0], .@item_id) != -1 ||
			inarray(.AllTreasures[0], .@item_id) == -1 ||
			getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_ETC)
			continue;
		.@treasure_group = callsub(S_GetTreasureGroup, .@item_id);
		if (.@treasure_group < 1 || .@treasure_group > 5)
			continue;
		.@amount = countitem(.@item_id);
		if (.@amount < 1)
			continue;
		.@bulk_ids[.@bulk_count] = .@item_id;
		.@bulk_amounts[.@bulk_count] = .@amount;
		.@bulk_groups[.@bulk_count++] = .@treasure_group;
		.@bulk_treasures += .@amount;
		.@group_counts[.@treasure_group - 1] += .@amount;
	}
	if (!.@bulk_count) {
		mes "[Moth]";
		mes "No snacks? Bring me an Iron, Silver, or Gold Treasure Box.";
		next;
		goto L_Main;
	}
	mes "[Moth]";
	mes "Choose Silvervine or Cookie. One plate, not both.";
	next;
	.@mode = select("Silvervine:Cookie:Keep every treasure.");
	clear;
	if (.@mode == 3)
		goto L_Main;
	.@reward_amount = 0;
	if (.@mode == 1) {
		.@reward_id = 6417;
		.@reward_name$ = "Silvervine Fruit";
		for (.@i = 0; .@i < 5; ++.@i)
			.@reward_amount += .@group_counts[.@i] * .FruitReward[.@i];
	} else {
		.@reward_id = 7000;
		.@reward_name$ = "Event Stone Coin";
		for (.@i = 0; .@i < 5; ++.@i)
			.@reward_amount += .@group_counts[.@i] * .CookieReward[.@i];
	}
	mes "[Moth]";
	mes "^0055FF" + .@bulk_count + " treasure types, " + .@bulk_treasures + " boxes total.^000000";
	next;
	for (.@i = 0; .@i < 5; ++.@i) {
		mes "[Moth]";
		mes .GroupLabels$[.@i] + ": " + .@group_counts[.@i] + " boxes.";
		next;
	}
	mes "[Moth]";
	mes "I will give you ^0055FF" + .@reward_amount + " " + .@reward_name$ + "^000000.";
	next;
	mes "[Moth]";
	mes "Once I crunch, there are no refunds.";
	if (select("Feed every accepted treasure.:Keep everything.") != 1)
		goto L_Main;
	clear;
	setarray .@reward_ids[0], .@reward_id;
	setarray .@reward_amounts[0], .@reward_amount;
	if (!checkweight2(.@reward_ids, .@reward_amounts)) {
		mes "[Moth]";
		mes "Your reward will not fit. Make room first. Nothing was eaten.";
		next;
		goto L_Main;
	}
	// Validate the complete snapshot before deleting the first box.
	for (.@i = 0; .@i < .@bulk_count; ++.@i) {
		if (getiteminfo(.@bulk_ids[.@i], ITEMINFO_TYPE) != IT_ETC ||
			inarray(.AllTreasures[0], .@bulk_ids[.@i]) == -1 ||
			callsub(S_GetTreasureGroup, .@bulk_ids[.@i]) != .@bulk_groups[.@i] ||
			countitem(.@bulk_ids[.@i]) < .@bulk_amounts[.@i]) {
			mes "[Moth]";
			mes "Something changed in your bag. Nothing was eaten.";
			next;
			goto L_Main;
		}
	}
	for (.@i = 0; .@i < .@bulk_count; ++.@i)
		delitem .@bulk_ids[.@i], .@bulk_amounts[.@i];
	getitem .@reward_id, .@reward_amount;
	logmes "Treasure Eater: bulk ate " + .@bulk_treasures + " boxes across " + .@bulk_count + " item IDs; groups " + .@group_counts[0] + "," + .@group_counts[1] + "," + .@group_counts[2] + "," + .@group_counts[3] + "," + .@group_counts[4] + "; mode " + .@reward_name$ + "; reward item " + .@reward_id + "; reward amount " + .@reward_amount;
	mes "[Moth]";
	mes "Crunch. Shiny outside, buggy inside. Your reward is ready.";
	next;
	goto L_Feed;

L_Feed:
	clear;
	getinventorylist;
	deletearray .@treasure_ids[0], getarraysize(.@treasure_ids);
	.@treasure_count = 0;
	for (.@i = 0; .@i < @inventorylist_count; ++.@i) {
		.@item_id = @inventorylist_id[.@i];
		if (inarray(.AllTreasures[0], .@item_id) == -1 ||
			getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_ETC)
			continue;
		if (inarray(.@treasure_ids[0], .@item_id) == -1)
			.@treasure_ids[.@treasure_count++] = .@item_id;
	}
	if (!.@treasure_count) {
		mes "[Moth]";
		mes "No snacks? Bring me an Iron, Silver, or Gold Treasure Box.";
		next;
		goto L_Main;
	}
	.@page = 0;
	while (true) {
		.@start = .@page * .PageSize;
		.@end = min(.@start + .PageSize, .@treasure_count);
		mes "[Moth]";
		mes "Pick my dinner by number. Page " + (.@page + 1) + " / " + ((.@treasure_count + .PageSize - 1) / .PageSize) + ".";
		next;
		for (.@i = .@start; .@i < .@end; ++.@i) {
			mes "[Moth]";
			mes "^0055FF" + (.@i + 1) + ".^000000 " + mesitemlink(.@treasure_ids[.@i], false) + " x" + countitem(.@treasure_ids[.@i]);
			if (.@i + 1 < .@end)
				next;
		}
		.@menu$ = "Choose a treasure.";
		.@next_option = 0;
		.@previous_option = 0;
		.@option = 2;
		if (.@end < .@treasure_count) {
			.@menu$ += ":Next page.";
			.@next_option = .@option++;
		}
		if (.@page > 0) {
			.@menu$ += ":Previous page.";
			.@previous_option = .@option++;
		}
		.@cancel_option = .@option;
		.@menu$ += ":Cancel.";
		.@action = select(.@menu$);
		clear;
		if (.@action == 1)
			break;
		if (.@action == .@next_option) {
			++.@page;
			continue;
		}
		if (.@action == .@previous_option) {
			--.@page;
			continue;
		}
		if (.@action == .@cancel_option)
			goto L_Main;
	}
	mes "[Moth]";
	mes "Enter the number shown beside the treasure.";
	input .@selection, .@start + 1, .@end;
	clear;
	.@selected_id = .@treasure_ids[.@selection - 1];
	.@owned = countitem(.@selected_id);
	if (.@owned < 1)
		goto L_Feed;
	mes "[Moth]";
	mes "You have " + .@owned + " " + mesitemlink(.@selected_id, false) + ".";
	next;
	mes "[Moth]";
	mes "How many boxes should I open for dinner?";
	input .@amount, 1, .@owned;
	clear;
	.@treasure_group = callsub(S_GetTreasureGroup, .@selected_id);
	if (.@treasure_group < 1 || .@treasure_group > 5)
		goto L_Main;
	.@silvervine_reward = .FruitReward[.@treasure_group - 1] * .@amount;
	.@cookie_reward = .CookieReward[.@treasure_group - 1] * .@amount;
	mes "[Moth]";
	mes "Choose Silvervine or Cookie. One plate, not both.";
	next;
	switch (select("Silvervine:Cookie:Keep the treasure.")) {
	case 1:
		.@reward_id = 6417;
		.@reward_amount = .@silvervine_reward;
		.@reward_name$ = "Silvervine Fruit";
		break;
	case 2:
		.@reward_id = 7000;
		.@reward_amount = .@cookie_reward;
		.@reward_name$ = "Event Stone Coin";
		break;
	default: goto L_Feed;
	}
	clear;
	mes "[Moth]";
	mes "^0055FF" + mesitemlink(.@selected_id, false) + " x" + .@amount + "^000000?";
	next;
	mes "[Moth]";
	mes "I will give you ^0055FF" + .@reward_amount + " " + .@reward_name$ + "^000000.";
	next;
	mes "[Moth]";
	mes "Once I crunch, there are no refunds.";
	if (select("Let the cat eat.:Keep the treasure.") != 1)
		goto L_Feed;
	clear;
	if (getiteminfo(.@selected_id, ITEMINFO_TYPE) != IT_ETC ||
		inarray(.AllTreasures[0], .@selected_id) == -1 ||
		countitem(.@selected_id) < .@amount ||
		callsub(S_GetTreasureGroup, .@selected_id) != .@treasure_group) {
		mes "[Moth]";
		mes "Something changed in your bag. Nothing was eaten.";
		next;
		goto L_Main;
	}
	setarray .@reward_ids[0], .@reward_id;
	setarray .@reward_amounts[0], .@reward_amount;
	if (!checkweight2(.@reward_ids, .@reward_amounts)) {
		mes "[Moth]";
		mes "Your reward will not fit. Make room first. Nothing was eaten.";
		next;
		goto L_Main;
	}
	delitem .@selected_id, .@amount;
	getitem .@reward_id, .@reward_amount;
	logmes "Treasure Eater: single ate " + .@amount + " x item " + .@selected_id + "; group " + .@treasure_group + "; mode " + .@reward_name$ + "; reward item " + .@reward_id + "; reward amount " + .@reward_amount;
	mes "[Moth]";
	mes "Crunch. Shiny outside, buggy inside. Your reward is ready.";
	next;
	goto L_Feed;

L_Catalog:
	clear;
	mes "[Moth]";
	mes "Pick a treasure group. I eat Iron, Silver, and Gold, not Diamond or Platinum.";
	next;
	.@catalog_group = select("Iron I-II.:Iron III-V.:Silver I-II.:Silver III-V.:Gold I-V.:Back.");
	if (.@catalog_group == 6)
		goto L_Main;
	deletearray .@catalog_ids[0], getarraysize(.@catalog_ids);
	switch (.@catalog_group) {
	case 1: copyarray .@catalog_ids[0], .Group1[0], getarraysize(.Group1); break;
	case 2: copyarray .@catalog_ids[0], .Group2[0], getarraysize(.Group2); break;
	case 3: copyarray .@catalog_ids[0], .Group3[0], getarraysize(.Group3); break;
	case 4: copyarray .@catalog_ids[0], .Group4[0], getarraysize(.Group4); break;
	case 5: copyarray .@catalog_ids[0], .Group5[0], getarraysize(.Group5); break;
	}
	for (.@i = 0; .@i < getarraysize(.@catalog_ids); ++.@i) {
		mes "[Moth]";
		mes mesitemlink(.@catalog_ids[.@i], false) + ": " + .FruitReward[.@catalog_group - 1] + " Silvervine or " + .CookieReward[.@catalog_group - 1] + " Cookie per box.";
		if (.@i + 1 < getarraysize(.@catalog_ids))
			next;
	}
	if (select("Back to groups.:Back to main.") == 1)
		goto L_Catalog;
	goto L_Main;

S_GetTreasureGroup:
	.@item_id = getarg(0);
	if (inarray(.Group1[0], .@item_id) != -1) return 1;
	if (inarray(.Group2[0], .@item_id) != -1) return 2;
	if (inarray(.Group3[0], .@item_id) != -1) return 3;
	if (inarray(.Group4[0], .@item_id) != -1) return 4;
	if (inarray(.Group5[0], .@item_id) != -1) return 5;
	return 0;

OnInit:
	freeloop 1;
	.PageSize = 10;
	setarray .FruitReward[0], 2,8,14,20,30;
	setarray .CookieReward[0], 10,30,50,60,90;
	setarray .GroupLabels$[0], "Iron I-II", "Iron III-V", "Silver I-II", "Silver III-V", "Gold I-V";
	setarray .Group1[0], 9000001,9000002;
	setarray .Group2[0], 9000003,9000004,9000005;
	setarray .Group3[0], 9000006,9000007;
	setarray .Group4[0], 9000008,9000009,9000010;
	setarray .Group5[0], 9000011,9000012,9000013,9000014,9000015;
	deletearray .AllTreasures[0], getarraysize(.AllTreasures);
	copyarray .AllTreasures[0], .Group1[0], getarraysize(.Group1);
	copyarray .AllTreasures[2], .Group2[0], getarraysize(.Group2);
	copyarray .AllTreasures[5], .Group3[0], getarraysize(.Group3);
	copyarray .AllTreasures[7], .Group4[0], getarraysize(.Group4);
	copyarray .AllTreasures[10], .Group5[0], getarraysize(.Group5);
	.Valid = 1;
	if (getarraysize(.Group1) != 2 || getarraysize(.Group2) != 3 || getarraysize(.Group3) != 2 || getarraysize(.Group4) != 3 || getarraysize(.Group5) != 5 || getarraysize(.AllTreasures) != 15) {
		debugmes "Treasure Eater: expected 15 boxes in groups 2/3/2/3/5.";
		.Valid = 0;
	}
	setarray .@expected_fruit[0], 2,8,14,20,30;
	setarray .@expected_cookie[0], 10,30,50,60,90;
	if (getarraysize(.FruitReward) != 5 || getarraysize(.CookieReward) != 5)
		.Valid = 0;
	for (.@i = 0; .@i < 5; ++.@i) {
		if (.FruitReward[.@i] != .@expected_fruit[.@i] || .CookieReward[.@i] != .@expected_cookie[.@i]) {
			debugmes "Treasure Eater: incorrect rewards for group " + (.@i + 1) + ".";
			.Valid = 0;
		}
	}
	for (.@i = 0; .@i < getarraysize(.AllTreasures); ++.@i) {
		.@item_id = .AllTreasures[.@i];
		if (.@item_id != 9000001 + .@i || getiteminfo(.@item_id, ITEMINFO_TYPE) != IT_ETC || inarray(.AllTreasures[0], .@item_id) != .@i) {
			debugmes "Treasure Eater: invalid or duplicate treasure item " + .@item_id + ".";
			.Valid = 0;
		}
	}
	if (!.Valid)
		debugmes "Treasure Eater: disabled interaction because menu validation failed.";
	else
		debugmes "Treasure Eater: loaded 15 treasure boxes (2/3/2/3/5).";
	freeloop 0;
	end;
}
