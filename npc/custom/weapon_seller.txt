//===== Ragnarok Offline =====================================
//= Black-Market Arms Importer
//= Soren imports proven iRO weapons from every direction.
//= Browse by weapon type; every current type opens its Equipment stock directly.
//= Every SKU uses the audited Cash Point price. Missing client assets do not
//= remove stock; this pass changes server data only.
//============================================================

paramk,51,175,2	script	Soren#offline	4_M_ROTERT,{
	if (!bmc_ready("soren")) {
		mes "[Soren]";
		mes "This stall's catalog is unavailable. Please try again later.";
		close;
	}
	.@eac = eaclass();
	if ((.@eac&EAJ_BASEMASK) == EAJ_SUMMONER) {
		mes "[Soren]";
		mes "A talking cat with opinions about weapons. Good.";
		mes "I import for discerning customers.";
		next;
	} else if (.@eac&EAJL_BABY) {
		mes "[Soren]";
		mes "Small frame. Serious stance. I know the difference.";
		mes "A weapon does not care how young its wielder looks.";
		next;
	} else if (.@eac&EAJL_UPPER) {
		mes "[Soren]";
		mes "You have the posture of someone who has survived old battles.";
		mes "Then you already know why good steel is worth finding.";
		next;
	}
	mes "[Soren]";
	mes "Weapons from every port, ruin, and battlefield I can reach.";
	mes "I trust what I import, and I will tell you when I do not.";
	mes "No boasting. No mystery eggs. Just proven weapons.";
	.@shop$ = callfunc("F_BlackMarketBrowse", "soren");
	if (.@shop$ == "") {
		clear;
		mes "[Soren]";
		mes "That crate is empty. Try another drawer.";
		close;
	}
	callfunc "F_BlackMarketDailyRemark", "Soren", $BlackMarketRateSoren;
	close2;
	callblackmarketshop .@shop$, 1, $BlackMarketRateSoren;
	end;

OnInit:
	setunittitle(getnpcid(0), "<Arms Importer>");
	end;
}
