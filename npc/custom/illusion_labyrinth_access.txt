function	script	F_IllusionLabyrinthTrace	{
	.@quest_id = getarg(0);
	.@member$ = getarg(1);
	.@npc$ = getarg(2);
	if (ill_laby != 3 || isbegin_quest(.@quest_id) != 1)
		return;

	mes "[Investigation]";
	mes "You find a faint trace left by " + .@member$ + ".";
	next;
	mes "[Investigation]";
	mes "It suggests that " + .@member$ + " passed through this distorted labyrinth, but it does not reveal where they are now.";
	next;
	mes "[Investigation]";
	mes "You record what you found so you can report it to Esmeralda.";
	completequest .@quest_id;
	cloakonnpcself( .@npc$ );

	if (isbegin_quest(12484) != 2 || isbegin_quest(12485) != 2 || isbegin_quest(12486) != 2 || isbegin_quest(12487) != 2) {
		next;
		mes "[Investigation]";
		mes "I should continue searching for traces of the other guild members.";
		close;
	}

	setquest 12488;
	ill_laby = 4;
	next;
	mes "[Investigation]";
	mes "I have recorded traces from all four guild members. I should report what I found to Esmeralda.";
	close2;
	warp "prt_maze01",99,26;
	end;
}

function	script	F_IllusionLabyrinthDailyReset	{
	if (checkquest(12490,PLAYTIME) == 2) {
		erasequest 12490;
		ill_laby_dq_done = 0;
		questinfo_refresh();
	}
	return;
}

function	script	F_IllusionLabyrinthWeeklyReset	{
	if (checkquest(12491,PLAYTIME) == 2) {
		erasequest 12491;
		questinfo_refresh();
	}
	return;
}

function	script	F_IllusionLabyrinthWeeklyWeek$	{
	.@day = gettime(DT_DAYOFWEEK);
	.@days_since_saturday = (.@day + 1) % 7;
	if (.@day == SATURDAY && gettime(DT_HOUR) < 4)
		.@days_since_saturday = 7;
	.@seconds_since_reset = .@days_since_saturday * 86400 + gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND) - 14400;
	return gettimestr("%Y%m%d",9,gettimetick(2) - .@seconds_since_reset);
}

			.@status$[.@i] = "Completed today";

			.@status$[.@i] = "Available";
		.@menu$ += (.@i == 0 ? "" : ":") + .@quest_titles$[.@i] + " [" + .@status$[.@i] + "]";
	}
	.@menu$ += ":Leave";
	.@choice = select(.@menu$);
