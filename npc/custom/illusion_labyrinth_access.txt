function	script	F_IllusionLabyrinthTrace	{
	.@quest_id = getarg(0);
	.@member$ = getarg(1);
	.@npc$ = getarg(2);
	if (ill_laby != 3 || isbegin_quest(.@quest_id) != 1)
		return;

	mes "[Investigation]";
	mes "You find a faint trace left by " + .@member$ + ".";
	next;
	mes "[Investigation]";
	mes "It suggests that " + .@member$ + " passed through this distorted labyrinth, but it does not reveal where they are now.";
	next;
	mes "[Investigation]";
	mes "You record what you found so you can report it to Esmeralda.";
	completequest .@quest_id;
	cloakonnpcself( .@npc$ );

	if (isbegin_quest(12484) != 2 || isbegin_quest(12485) != 2 || isbegin_quest(12486) != 2 || isbegin_quest(12487) != 2) {
		next;
		mes "[Investigation]";
		mes "I should continue searching for traces of the other guild members.";
		close;
	}

	setquest 12488;
	ill_laby = 4;
	next;
	mes "[Investigation]";
	mes "I have recorded traces from all four guild members. I should report what I found to Esmeralda.";
	close2;
	warp "prt_maze01",99,26;
	end;
}
