function	script	F_IllusionLabyrinthTrace	{
	.@quest = getarg(0); .@bit = getarg(1); .@npc$ = getarg(2);
	if (ill_laby != 3 || (ill_laby_trace_mask & .@bit)) return;
	mes "A faint trace of a Priest Guild member remains here.";
	completequest .@quest;
	ill_laby_trace_mask |= .@bit;
	ill_laby_trace_count++;
	cloakonnpcself( .@npc$ );
	if (ill_laby_trace_count < 3) { mes "I should continue searching for the others."; close; }
	if (!(ill_laby_trace_mask & 1)) erasequest 12484;
	if (!(ill_laby_trace_mask & 2)) erasequest 12485;
	if (!(ill_laby_trace_mask & 4)) erasequest 12486;
	if (!(ill_laby_trace_mask & 8)) erasequest 12487;
	setquest 12488;
	ill_laby = 4;
	mes "I have enough evidence. I should report to Esmeralda.";
	close2;
	warp "prt_maze01",101,26;
	end;
}
