//===== Ragnarok Offline =====================================
//= Premium Buff and Premium Service Staff
//===== Description: =========================================
//= Timed account Premium Subscription (rAthena VIP). Duration
//= boxes grant a rental Premium Service Staff from box minutes
//= plus remaining time (do not wait for async vip_time ack).
//= GM @vip <seconds> applies to self; @vip <char_name> <seconds>
//= adds seconds; 0 clears VIP, #PREMIUM_UNTIL, AID icons, and the rental wand.
//= #vip is disabled. Player group cannot use @vip.
//= Using the wand duplicates Premi next to the player.
//= Premi#src must sit on a real map cell (izlude 1,1). A floating
//= "-" source becomes JT_FAKENPC and duplicate_dynamic never spawns.
//= Premi and AID icons follow #PREMIUM_UNTIL, not vip_status.
//= Extra 24-hour options cost Nyangvine.
//= Sprite 4_F_PREMI (10209) with official illust prm_1-5.
//= Do not use 4_SERVICE_30_* (different black-outfit cats)
//= or 4_CAT_ADMIRAL. Do not name her ABC (Abyss Chaser).
//============================================================

function	script	F_PremiumRemain	{
	.@expire = getarg(0);
	.@now = gettimetick(2);
	if (.@expire <= .@now)
		return "inactive";
	.@left = .@expire - .@now;
	return (.@left / 3600) + "h " + ((.@left % 3600) / 60) + "m";
}

// sc_start duration is int32 milliseconds; 30 days overflows.
function	script	F_PremiumApplyIcons	{
	.@seconds = getarg(0);
	if (.@seconds <= 0)
		return;
	if (.@seconds > 2147483)
		.@ms = 2147483647;
	else
		.@ms = .@seconds * 1000;
	sc_start SC_AID_PERIOD_PLUSEXP, .@ms, 20;
	sc_start SC_AID_PERIOD_RECEIVEITEM, .@ms, 20;
	return;
}

function	script	F_PremiumRemainingSeconds	{
	.@now = gettimetick(2);
	.@remain = vip_status(VIP_STATUS_REMAINING);
	if (.@remain < 0)
		.@remain = 0;
	if (#PREMIUM_UNTIL > .@now) {
		.@left = #PREMIUM_UNTIL - .@now;
		if (.@left > .@remain)
			.@remain = .@left;
	}
	return .@remain;
}

function	script	F_PremiumResyncStaff	{
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (countitem(23165) > 0)
		delitem 23165, countitem(23165);
	if (rentalcountitem(23165) > 0)
		delitem 23165, rentalcountitem(23165);
	if (.@remain <= 0) {
		#PREMIUM_UNTIL = 0;
		sc_end SC_AID_PERIOD_PLUSEXP;
		sc_end SC_AID_PERIOD_RECEIVEITEM;
		return;
	}
	callfunc("F_PremiumApplyIcons", .@remain);
	rentitem 23165, .@remain;
	return;
}

function	script	F_PremiumSyncStaff	{
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (.@remain <= 0)
		return;
	callfunc("F_PremiumApplyIcons", .@remain);
	if (rentalcountitem(23165) < 1) {
		if (countitem(23165) > 0)
			delitem 23165, countitem(23165);
		rentitem 23165, .@remain;
	}
	return;
}

function	script	F_PremiumSubscription	{
	.@minutes = getarg(0);
	.@old = callfunc("F_PremiumRemainingSeconds");
	.@seconds = .@old + (.@minutes * 60);
	#PREMIUM_UNTIL = gettimetick(2) + .@seconds;
	vip_time .@minutes;
	if (countitem(23165) > 0)
		delitem 23165, countitem(23165);
	if (rentalcountitem(23165) > 0)
		delitem 23165, rentalcountitem(23165);
	rentitem 23165, .@seconds;
	callfunc("F_PremiumApplyIcons", .@seconds);
	.@days = .@minutes / 1440;
	if (.@days == 1)
		dispbottom "Premium Buff activated for 1 day.";
	else
		dispbottom "Premium Buff activated for " + .@days + " days.";
	return;
}

function	script	F_PremiumServiceStaff	{
	.@now = gettimetick(2);
	if (#PREMIUM_STAFF_CD > .@now)
		return;
	#PREMIUM_STAFF_CD = .@now + 3;
	if (getmapflag(strcharinfo(3), mf_nodynamicnpc)) {
		dispbottom "The Premium Service Staff cannot be called on this map.";
		return;
	}
	.@dup$ = duplicate_dynamic("Premi#src");
	if (.@dup$ == "") {
		dispbottom "Premi is already nearby, or there is no open cell. Move a few steps and try again.";
		return;
	}
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (.@remain > 0)
		callfunc("F_PremiumApplyIcons", .@remain);
	callfunc("F_PremiumSyncExtras");
	dispbottom "Premi appears nearby.";
	return;
}

function	script	F_PremiumExtraRunning	{
	.@kind = getarg(0);
	.@now = gettimetick(2);
	.@exp = (#PREMIUM_EXPUP > .@now || getstatus(SC_PERIOD_PLUSEXP_2ND));
	.@item = (#PREMIUM_ITEMUP > .@now || getstatus(SC_PERIOD_RECEIVEITEM_2ND));
	.@death = (#PREMIUM_DEATHDOWN > .@now || getstatus(SC_LIFEINSURANCE));
	.@card = (#PREMIUM_CARDLOOT > .@now);
	if (.@kind == 1)
		return .@exp;
	if (.@kind == 2)
		return .@item;
	if (.@kind == 3)
		return .@death;
	if (.@kind == 4)
		return .@card;
	if (.@kind == 5)
		return (.@exp || .@item || .@death || .@card);
	return 0;
}

// Keep Card Magnet account-wide without creating a new SC_* enum. A nonzero
// expired stamp is deliberately cleared here; a zero stamp must not disturb a
// player's independent @aloottype +card choice.
function	script	F_PremiumSyncCardLoot	{
	deltimer "PremiumLogin#offline::OnCardLootExpire";
	.@now = gettimetick(2);
	if (#PREMIUM_CARDLOOT <= .@now) {
		if (#PREMIUM_CARDLOOT > 0) {
			#PREMIUM_CARDLOOT = 0;
			premiumcardloot 0,0;
		}
		return 0;
	}
	.@ms = (#PREMIUM_CARDLOOT - .@now) * 1000;
	if (premiumcardloot(1, .@ms) == 0)
		return 0;
	addtimer .@ms, "PremiumLogin#offline::OnCardLootExpire";
	return 1;
}

function	script	F_PremiumApplyExtra	{
	.@kind = getarg(0);
	if (callfunc("F_PremiumExtraRunning", .@kind))
		return 0;
	.@now = gettimetick(2);
	.@until = .@now + 86400;
	if (.@kind == 4 || .@kind == 5) {
		if (premiumcardloot(1, 86400000) == 0)
			return 0;
		#PREMIUM_CARDLOOT = .@until;
	}
	if (.@kind == 1 || .@kind == 5) {
		#PREMIUM_EXPUP = .@until;
		sc_start SC_PERIOD_PLUSEXP_2ND, 86400000, 30;
	}
	if (.@kind == 2 || .@kind == 5) {
		#PREMIUM_ITEMUP = .@until;
		sc_start SC_PERIOD_RECEIVEITEM_2ND, 86400000, 30;
	}
	if (.@kind == 3 || .@kind == 5) {
		#PREMIUM_DEATHDOWN = .@until;
		sc_start SC_LIFEINSURANCE, 86400000, 0;
	}
	if (.@kind == 4 || .@kind == 5)
		callfunc("F_PremiumSyncCardLoot");
	return 1;
}

function	script	F_PremiumSyncExtras	{
	.@now = gettimetick(2);
	if (#PREMIUM_EXPUP > .@now)
		sc_start SC_PERIOD_PLUSEXP_2ND, (#PREMIUM_EXPUP - .@now) * 1000, 30;
	if (#PREMIUM_ITEMUP > .@now)
		sc_start SC_PERIOD_RECEIVEITEM_2ND, (#PREMIUM_ITEMUP - .@now) * 1000, 30;
	if (#PREMIUM_DEATHDOWN > .@now)
		sc_start SC_LIFEINSURANCE, (#PREMIUM_DEATHDOWN - .@now) * 1000, 0;
	callfunc("F_PremiumSyncCardLoot");
	return;
}

// Return the classic Soul Link skill that applies to this character's lineage.
// A zero result intentionally means that Premi still sells Package 5, but only
// gives Packages 1-4 and scolds the player for choosing a linkless job.
function	script	F_PremiumSoulLinkSkill	{
	.@eac = eaclass();
	.@base = .@eac&EAJ_BASEMASK;

	// Spirit of Rebirth is limited by rAthena to High First Classes below Lv. 70.
	if ((.@eac&EAJL_UPPER) && !(.@eac&EAJL_2) && !(.@eac&EAJL_THIRD) && !(.@eac&EAJL_FOURTH) && BaseLevel < 70) {
		if (.@base == EAJ_SWORDMAN || .@base == EAJ_MAGE || .@base == EAJ_ARCHER || .@base == EAJ_ACOLYTE || .@base == EAJ_MERCHANT || .@base == EAJ_THIEF)
			return 494; // SL_HIGH / Spirit of Rebirth
	}

	// EAJ_UPPERMASK is only enough for second/trans-second jobs. Third and
	// Fourth jobs retain their own map IDs there, so resolve their lineage from
	// EAJ_THIRDMASK first (it strips the Fourth-job bit as well).
	switch (.@eac&EAJ_THIRDMASK) {
	case EAJ_RUNE_KNIGHT: return 452; // SL_KNIGHT
	case EAJ_WARLOCK: return 453; // SL_WIZARD
	case EAJ_RANGER: return 460; // SL_HUNTER
	case EAJ_ARCH_BISHOP: return 454; // SL_PRIEST, including Cardinal
	case EAJ_MECHANIC: return 458; // SL_BLACKSMITH
	case EAJ_GUILLOTINE_CROSS: return 457; // SL_ASSASIN
	case EAJ_ROYAL_GUARD: return 450; // SL_CRUSADER
	case EAJ_SORCERER: return 449; // SL_SAGE
	case EAJ_MINSTRELWANDERER: return 455; // SL_BARDDANCER
	case EAJ_SURA: return 447; // SL_MONK
	case EAJ_GENETIC: return 445; // SL_ALCHEMIST
	case EAJ_SHADOW_CHASER: return 456; // SL_ROGUE
	case EAJ_SOUL_REAPER: return 461; // SL_SOULLINKER
	case EAJ_STAR_EMPEROR: return 448; // SL_STAR
	case EAJ_SUPER_NOVICE_E: return 451; // SL_SUPERNOVICE
	}

	switch (.@eac&EAJ_UPPERMASK) {
	// Skill names are not script constants. Keep these verified skill_db IDs.
	case EAJ_ALCHEMIST: return 445; // SL_ALCHEMIST
	case EAJ_ASSASSIN: return 457; // SL_ASSASIN
	case EAJ_BARDDANCER: return 455; // SL_BARDDANCER
	case EAJ_BLACKSMITH: return 458; // SL_BLACKSMITH
	case EAJ_CRUSADER: return 450; // SL_CRUSADER
	case EAJ_HUNTER: return 460; // SL_HUNTER
	case EAJ_KNIGHT: return 452; // SL_KNIGHT
	case EAJ_MONK: return 447; // SL_MONK
	case EAJ_PRIEST: return 454; // SL_PRIEST, including High Priest
	case EAJ_ROGUE: return 456; // SL_ROGUE
	case EAJ_SAGE: return 449; // SL_SAGE
	case EAJ_SOULLINKER: return 461; // SL_SOULLINKER
	case EAJ_STARGLADIATOR: return 448; // SL_STAR
	case EAJ_SUPERNOVICE: return 451; // SL_SUPERNOVICE
	case EAJ_WIZARD: return 453; // SL_WIZARD
	}
	return 0;
}

OnCardLootExpire:
	.@now = gettimetick(2);

		end;
	}

		premiumcardloot 0,0;
		dispbottom "Card Magnet expired.";
	}
	end;

	end;

	if (.@remain > 0)
		callfunc("F_PremiumApplyIcons", .@remain);

		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah-- another cat?";
		mes "I-I'm working, okay?";
		next;
		cutin "prm_2",2;
	} else if (.@eac&EAJL_BABY) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "You don't feel like a kitten...";
		mes "Ehehe. I can still help!";
		next;
		cutin "prm_2",2;
	}
	mes "[Premi]";
	mes "Hi hi! Premi is here!";
	next;
	if (.@remain <= 0) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "Aww... your Premium Buff isn't on.";
		mes "Open a Premium Subscription box first, okay?";
		close2;
		cutin "",255;
		end;
	}
	cutin "prm_1",2;
	mes "[Premi]";
	mes "Your Premium Buff is still going until:";
	mes gettimestr("%Y-%m/%d %H:%M:%S", 21, gettimetick(2) + .@remain);
	next;
	cutin "prm_1",2;
	mes "[Premi]";
	mes "What do you need?";
	next;
	.@in_black_market = (strcharinfo(3) == "paramk");
	.@menu$ = "Buff me.:Premium extras.:";
	if (!.@in_black_market)
		.@menu$ += "Take me to the Black Market. (1 Nyangvine):";
	.@menu$ += "Just checking:";
	.@choice = select(.@menu$);
	if (.@choice == 1) {
		goto S_BuffMenu;
	}
	if (.@choice == 2) {
		goto S_ExtrasMenu;
	}
	if (!.@in_black_market && .@choice == 3) {
		goto S_BlackMarket;
	}
	cutin "prm_3",2;
	mes "[Premi]";
	mes "Okay! Call me again if you need anything!";
	goto L_End;

S_BlackMarket:
	// Premi is duplicated onto the invoking player's current map. With no mode
	// argument, instance_id() therefore checks that map rather than an owned
	// party/guild/character instance that the player may not currently be in.
	if (instance_id() > 0) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "I can't take you to the Black Market from an instance map...";
		mes "Finish your instance first, okay?";
		goto L_End;
	}
	if (countitem(6909) < 1) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "I need 1 Nyangvine to get you into the Black Market...";
		goto L_End;
	}
	cutin "prm_2",2;
	mes "[Premi]";
	mes "For 1 Nyangvine, I'll remember exactly where you're standing.";
	mes "The Black Market Contact can bring you back here afterward!";
	next;
	if (select("Travel:Never mind") == 2) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
		goto L_End;
	}
	if (instance_id() > 0) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "I can't take you to the Black Market from an instance map...";
		goto L_End;
	}
	if (countitem(6909) < 1) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "The Nyangvine is gone...";
		goto L_End;
	}
	if (getmapxy(.@return_map$,.@return_x,.@return_y) != 0) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "I couldn't remember where you're standing. Try calling me again.";
		goto L_End;
	}
	// Permanent character variables. Marker 12 selects this exact-position
	// return path in Black Market Contact#exit.
	BM_ReturnMap$ = .@return_map$;
	BM_ReturnX = .@return_x;
	BM_ReturnY = .@return_y;
	BM_ReturnLocation = 12;
	delitem 6909,1;
	cutin "prm_3",2;
	mes "[Premi]";
	mes "Okay! Stay close!";
	close2;
	cutin "",255;
	warp "paramk",11,171;
	end;

S_BuffMenu:
	cutin "prm_2",2;
	mes "[Premi]";
	mes "All package buffs last 5 minutes, just for this character!";
	next;
	.@buff_menu$ = "Package 1 - AGI, Blessing, Angelus (2 Nyangvine):"
		+ "Package 2 - Package 1 + Magnificat, Kyrie (3 Nyangvine):"
		+ "Package 3 - Package 2 + Impositio, Assumptio, Suffragium (5 Nyangvine):"
		+ "Package 4 - Package 3 + Wind Walk, Power-Thrust, Weapon Perfection (7 Nyangvine):"
		+ "Package 5 - Package 4 + your Soul Link (10 Nyangvine):"
		+ "Never mind:";
	switch (select(.@buff_menu$)) {
	case 1:
		callsub S_BuyBuff, 1, 2, "Package 1";
		break;
	case 2:
		callsub S_BuyBuff, 2, 3, "Package 2";
		break;
	case 3:
		callsub S_BuyBuff, 3, 5, "Package 3";
		break;
	case 4:
		callsub S_BuyBuff, 4, 7, "Package 4";
		break;
	case 5:
		callsub S_BuyBuff, 5, 10, "Package 5";
		break;
	default:
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
	}
	goto L_End;

S_ExtrasMenu:
	cutin "prm_1",2;
	mes "[Premi]";
	mes "Extra options last 24 hours for every character on this account!";
	mes "They cost Nyangvine, and I won't sell one that's already running.";
	next;
	.@menu$ = "EXP UP (4 Nyangvine):ITEM UP (2 Nyangvine):DEATH DOWN (1 Nyangvine):CARD MAGNET (1 Nyangvine):Advanced Package (8 Nyangvine):Just checking:";
	switch (select(.@menu$)) {
	case 1:
		callsub S_RefuseIfRunning, 1;
		callsub S_Buy, 4, 1, "EXP UP";
		end;
	case 2:
		callsub S_RefuseIfRunning, 2;
		callsub S_Buy, 2, 2, "ITEM UP";
		end;
	case 3:
		callsub S_RefuseIfRunning, 3;
		callsub S_Buy, 1, 3, "DEATH DOWN";
		end;
	case 4:
		callsub S_RefuseIfRunning, 4;
		callsub S_Buy, 1, 4, "Card Magnet";
		end;
	case 5:
		callsub S_RefuseIfRunning, 5;
		callsub S_Buy, 8, 5, "Advanced Package";
		end;
	default:
		cutin "prm_3",2;
		mes "[Premi]";
		mes "Okay! Call me again if you need anything!";
		goto L_End;
	}

S_BuyBuff:
	.@tier = getarg(0);
	.@cost = getarg(1);
	.@label$ = getarg(2);
	// Resolve once here so the paid Package 5 result and its dialogue use
	// exactly the same Soul Link decision.
	.@spirit = 0;
	if (.@tier == 5)
		.@spirit = callfunc("F_PremiumSoulLinkSkill");
	if (countitem(6909) < .@cost) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "That's " + .@cost + " Nyangvine for " + .@label$ + "...";
		mes "You don't have enough. Dang it...";
		return;
	}
	cutin "prm_2",2;
	mes "[Premi]";
	mes .@label$ + " costs " + .@cost + " Nyangvine for 5 minutes.";
	next;
	if (select("Pay:Never mind") == 2) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
		return;
	}
	if (countitem(6909) < .@cost) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "The Nyangvine is gone...";
		return;
	}
	delitem 6909, .@cost;
	callsub S_ApplyBuffPackage, .@tier, .@spirit;
	cutin "prm_3",2;
	mes "[Premi]";
	mes "Yay! " + .@label$ + " is running for 5 minutes!";
	if (.@tier == 5 && !.@spirit) {
		cutin "prm_4",2;
		mes "Your job doesn't have a classic Soul Link...";
		mes "You paid for Package 5 without checking first. That's on you!";
	}
	return;

S_ApplyBuffPackage:
	.@tier = getarg(0);
	.@spirit = getarg(1);
	.@buff_time = 300000;
	sc_start SC_INCREASEAGI, .@buff_time, 10;
	sc_start SC_BLESSING, .@buff_time, 10;
	sc_start SC_ANGELUS, .@buff_time, 10;
	if (.@tier >= 2) {
		sc_start SC_MAGNIFICAT, .@buff_time, 5;
		sc_start SC_KYRIE, .@buff_time, 10;
	}
	if (.@tier >= 3) {
		sc_start SC_IMPOSITIO, .@buff_time, 5;
		sc_start SC_ASSUMPTIO, .@buff_time, 5;
		sc_start SC_SUFFRAGIUM, .@buff_time, 3;
	}
	if (.@tier >= 4) {
		sc_start SC_WINDWALK, .@buff_time, 10;
		// val2 = 1 gives the self-cast Power-Thrust Lv.5 effect.
		sc_start2 SC_OVERTHRUST, .@buff_time, 5, 1;
		sc_start SC_WEAPONPERFECTION, .@buff_time, 5;
	}
	if (.@tier == 5 && .@spirit)
		sc_start2 SC_SPIRIT, .@buff_time, 5, .@spirit;
	return;

S_RefuseIfRunning:
	.@kind = getarg(0);
	if (!callfunc("F_PremiumExtraRunning", .@kind))
		return;
	cutin "prm_4",2;
	mes "[Premi]";
	if (.@kind == 5) {
		mes "I can't sell extra options while one is already running...";
	} else {
		if (.@kind == 1)
			.@left$ = callfunc("F_PremiumRemain", #PREMIUM_EXPUP);
		else if (.@kind == 2)
			.@left$ = callfunc("F_PremiumRemain", #PREMIUM_ITEMUP);
		else if (.@kind == 3)
			.@left$ = callfunc("F_PremiumRemain", #PREMIUM_DEATHDOWN);
		else
			.@left$ = callfunc("F_PremiumRemain", #PREMIUM_CARDLOOT);
		if (.@left$ == "inactive")
			mes "That extra is already running...";
		else
			mes "That extra is already running (" + .@left$ + ")...";
	}
	mes "I can't stack it. Sorry...";
	goto L_End;

S_Buy:
	.@cost = getarg(0);
	.@kind = getarg(1);
	.@label$ = getarg(2);
	if (callfunc("F_PremiumExtraRunning", .@kind)) {
		callsub S_RefuseIfRunning, .@kind;
		end;
	}
	if (countitem(6909) < .@cost) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "That's " + .@cost + " Nyangvine for " + .@label$ + "...";
		mes "You don't have enough. Dang it...";
		goto L_End;
	}
	cutin "prm_2",2;
	mes "[Premi]";
	mes .@label$ + " for 24 hours! " + .@cost + " Nyangvine, okay?";
	next;
	if (select("Pay:Never mind") == 2) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
		goto L_End;
	}
	if (callfunc("F_PremiumExtraRunning", .@kind)) {
		callsub S_RefuseIfRunning, .@kind;
		end;
	}
	if (countitem(6909) < .@cost) {
		cutin "prm_4",2;
		mes "[Premi]";
		mes "The Nyangvine is gone...";
		goto L_End;
	}
	delitem 6909, .@cost;
	if (callfunc("F_PremiumApplyExtra", .@kind) == 0) {
		getitem 6909, .@cost;
		callsub S_RefuseIfRunning, .@kind;
		end;
	}
	cutin "prm_3",2;
	mes "[Premi]";
	mes "Yay! " + .@label$ + " is running!";
	goto L_End;

L_End:
	close2;
	cutin "",255;
	end;
