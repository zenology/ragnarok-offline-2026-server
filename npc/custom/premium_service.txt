//===== Ragnarok Offline =====================================
//= Premium Buff and Premium Service Staff
//===== Description: =========================================
//= Timed account Premium Subscription (rAthena VIP). Duration
//= boxes grant a rental Premium Service Staff from box minutes
//= plus remaining time (do not wait for async vip_time ack).
//= GM @vip <seconds> applies to self; @vip <char_name> <seconds>
//= adds seconds; 0 clears VIP, #PREMIUM_UNTIL, AID icons, and the rental wand.
//= #vip is disabled. Player group cannot use @vip.
//= Using the wand duplicates Premi through the custom premisummon wrapper.
//= Premi-only Teleport effects and paid Kafra Storage live in src/custom.
//= Premi#src must sit on a real map cell (izlude 1,1). A floating
//= "-" source becomes JT_FAKENPC and duplicate_dynamic never spawns.
//= Premi and AID icons follow #PREMIUM_UNTIL, not vip_status.
//= Extra 24-hour options cost Nyangvine.
//= Sprite 4_F_PREMI (10209) with official illust prm_1-5.
//= Do not use 4_SERVICE_30_* (different black-outfit cats)
//= or 4_CAT_ADMIRAL. Do not name her ABC (Abyss Chaser).
//============================================================

function	script	F_PremiumRemain	{
	.@expire = getarg(0);
	.@now = gettimetick(2);
	if (.@expire <= .@now)
		return "inactive";
	.@left = .@expire - .@now;
	return (.@left / 3600) + "h " + ((.@left % 3600) / 60) + "m";
}

// sc_start duration is int32 milliseconds; 30 days overflows.
function	script	F_PremiumApplyIcons	{
	.@seconds = getarg(0);
	if (.@seconds <= 0)
		return;
	if (.@seconds > 2147483)
		.@ms = 2147483647;
	else
		.@ms = .@seconds * 1000;
	sc_start SC_AID_PERIOD_PLUSEXP, .@ms, 20;
	sc_start SC_AID_PERIOD_RECEIVEITEM, .@ms, 20;
	return;
}

function	script	F_PremiumRemainingSeconds	{
	.@now = gettimetick(2);
	.@remain = vip_status(VIP_STATUS_REMAINING);
	if (.@remain < 0)
		.@remain = 0;
	if (#PREMIUM_UNTIL > .@now) {
		.@left = #PREMIUM_UNTIL - .@now;
		if (.@left > .@remain)
			.@remain = .@left;
	}
	return .@remain;
}

function	script	F_PremiumResyncStaff	{
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (countitem(23165) > 0)
		delitem 23165, countitem(23165);
	if (rentalcountitem(23165) > 0)
		delitem 23165, rentalcountitem(23165);
	if (.@remain <= 0) {
		#PREMIUM_UNTIL = 0;
		sc_end SC_AUTOBUFF;
		sc_end SC_AID_PERIOD_PLUSEXP;
		sc_end SC_AID_PERIOD_RECEIVEITEM;
		return;
	}
	callfunc("F_PremiumApplyIcons", .@remain);
	rentitem 23165, .@remain;
	return;
}

function	script	F_PremiumSyncStaff	{
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (.@remain <= 0)
		return;
	callfunc("F_PremiumApplyIcons", .@remain);
	if (rentalcountitem(23165) < 1) {
		if (countitem(23165) > 0)
			delitem 23165, countitem(23165);
		rentitem 23165, .@remain;
	}
	return;
}

function	script	F_PremiumSubscription	{
	.@minutes = getarg(0);
	.@old = callfunc("F_PremiumRemainingSeconds");
	.@seconds = .@old + (.@minutes * 60);
	#PREMIUM_UNTIL = gettimetick(2) + .@seconds;
	vip_time .@minutes;
	if (countitem(23165) > 0)
		delitem 23165, countitem(23165);
	if (rentalcountitem(23165) > 0)
		delitem 23165, rentalcountitem(23165);
	rentitem 23165, .@seconds;
	callfunc("F_PremiumApplyIcons", .@seconds);
	.@days = .@minutes / 1440;
	if (.@days == 1)
		dispbottom "Premium Buff activated for 1 day.";
	else
		dispbottom "Premium Buff activated for " + .@days + " days.";
	return;
}

function	script	F_PremiumServiceStaff	{
	.@now = gettimetick(2);
	if (#PREMIUM_STAFF_CD > .@now)
		return;
	#PREMIUM_STAFF_CD = .@now + 3;
	if (getmapflag(strcharinfo(3), mf_nodynamicnpc)) {
		dispbottom "The Premium Service Staff cannot be called on this map.";
		return;
	}
	.@dup$ = premisummon();
	if (.@dup$ == "") {
		dispbottom "Premi is already nearby, or there is no open cell. Move a few steps and try again.";
		return;
	}
	.@remain = callfunc("F_PremiumRemainingSeconds");
	if (.@remain > 0)
		callfunc("F_PremiumApplyIcons", .@remain);
	callfunc("F_PremiumSyncExtras");
	dispbottom "Premi appears nearby.";
	return;
}

// Keep Treasure Magnet account-wide without creating a new SC_* enum.
// Retain #PREMIUM_CARDLOOT as its expiry key so previously purchased time is preserved.
// This service never changes the player's independent autoloot settings.
function	script	F_PremiumSyncTreasureLoot	{
	deltimer "PremiumLogin#offline::OnTreasureLootExpire";
	.@now = gettimetick(2);
	if (#PREMIUM_CARDLOOT <= .@now) {
		if (#PREMIUM_CARDLOOT > 0) {
			#PREMIUM_CARDLOOT = 0;
		}
		premiumtreasureloot 0,0;
		return 0;
	}
	.@ms = (#PREMIUM_CARDLOOT - .@now) * 1000;
	if (premiumtreasureloot(1, .@ms) == 0)
		return 0;
	addtimer .@ms, "PremiumLogin#offline::OnTreasureLootExpire";
	return 1;
}

function	script	F_PremiumSyncDropPenalty	{
	deltimer "PremiumLogin#offline::OnDropPenaltyExpire";
	.@now = gettimetick(2);
	if (#PREMIUM_NODROPPENALTY <= .@now) {
		#PREMIUM_NODROPPENALTY = 0;
		premiumdroppenaltyicon;
		return;
	}
	premiumdroppenaltyicon;
	addtimer (#PREMIUM_NODROPPENALTY - .@now) * 1000, "PremiumLogin#offline::OnDropPenaltyExpire";
	return;
}

// The caller validates payment before entering this non-yielding delivery block.
// Only selected extras get fresh time; unselected account deadlines/statuses stay intact.
function	script	F_PremiumApplyExtras	{
	.@mask = getarg(0);
	.@now = gettimetick(2);
	.@until = .@now + 86400;
	if (.@mask & 1) {
		sc_end SC_PERIOD_PLUSEXP_2ND;
		#PREMIUM_EXPUP = .@until;
		sc_start SC_PERIOD_PLUSEXP_2ND, 86400000, 30;
	}
	if (.@mask & 2) {
		sc_end SC_PERIOD_RECEIVEITEM_2ND;
		#PREMIUM_ITEMUP = .@until;
		sc_start SC_PERIOD_RECEIVEITEM_2ND, 86400000, 30;
	}
	if (.@mask & 4) {
		sc_end SC_LIFEINSURANCE;
		#PREMIUM_DEATHDOWN = .@until;
		sc_start SC_LIFEINSURANCE, 86400000, 0;
	}
	if (.@mask & 8) {
		#PREMIUM_CARDLOOT = .@until;
		callfunc("F_PremiumSyncTreasureLoot");
	}
	if (.@mask & 16) {
		#PREMIUM_NODROPPENALTY = .@until;
		callfunc("F_PremiumSyncDropPenalty");
	}
	return;
}

function	script	F_PremiumSyncExtras	{
	.@now = gettimetick(2);
	if (#PREMIUM_EXPUP > .@now)
		sc_start SC_PERIOD_PLUSEXP_2ND, (#PREMIUM_EXPUP - .@now) * 1000, 30;
	if (#PREMIUM_ITEMUP > .@now)
		sc_start SC_PERIOD_RECEIVEITEM_2ND, (#PREMIUM_ITEMUP - .@now) * 1000, 30;
	if (#PREMIUM_DEATHDOWN > .@now)
		sc_start SC_LIFEINSURANCE, (#PREMIUM_DEATHDOWN - .@now) * 1000, 0;
	callfunc("F_PremiumSyncTreasureLoot");
	callfunc("F_PremiumSyncDropPenalty");
	return;
}

-	script	PremiumLogin#offline	-1,{
	end;

OnPCLoginEvent:
	callfunc("F_PremiumSyncStaff");
	callfunc("F_PremiumSyncExtras");
	end;

OnTreasureLootExpire:
	.@now = gettimetick(2);
	if (#PREMIUM_CARDLOOT > .@now) {
		callfunc("F_PremiumSyncTreasureLoot");
		end;
	}
	if (#PREMIUM_CARDLOOT > 0) {
		#PREMIUM_CARDLOOT = 0;
		dispbottom "Treasure Magnet expired.";
	}
	premiumtreasureloot 0,0;
	end;

OnGmVipSync:
	callfunc("F_PremiumResyncStaff");
	end;

OnDropPenaltyExpire:
	if (#PREMIUM_NODROPPENALTY > gettimetick(2)) {
		callfunc("F_PremiumSyncDropPenalty");
		end;
	}
	if (#PREMIUM_NODROPPENALTY > 0)
		dispbottom "Ignore Item Drop Penalty expired.";
	#PREMIUM_NODROPPENALTY = 0;
	premiumdroppenaltyicon;
	end;
}

		goto L_End;

		mes "[Premi]";
		mes "I need 1 Nyangvine to open your Storage.";
		goto L_End;
	}
	mes "[Premi]";
	mes "I'll open your Kafra Storage for 1 Nyangvine!";

		goto L_End;

		mes "[Premi]";
		mes "You need Basic Skill Level 6 to use Storage.";

	}

	// argument, instance_id() therefore checks that map rather than an owned
	// party/guild/character instance that the player may not currently be in.
	if (instance_id() > 0) {
		cutin "prm_4",2;

		mes "I can't take you to the Black Market from an instance map...";
		mes "Finish your instance first, okay?";

		cutin "prm_4",2;
		mes "[Premi]";
		mes "I need 1 Nyangvine to get you into the Black Market...";
		goto L_End;

	cutin "prm_2",2;
	mes "[Premi]";
	mes "For 1 Nyangvine, I'll remember exactly where you're standing.";
	mes "The Black Market Contact can bring you back here afterward!";

	if (select("Travel:Never mind") == 2) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
		goto L_End;
	}

		cutin "prm_4",2;
		mes "[Premi]";

		mes "The Nyangvine is gone...";

		goto L_End;

	BM_ReturnX = .@return_x;

	mes "Okay! Stay close!";

	BM_Visited = 1;
	end;

		mes "[Premi]";
		mes "Auto Buff is already active. Cancel it before buying another package.";
		goto L_End;
	}
	cutin "prm_2",2;
	mes "[Premi]";
	mes "All package buffs last 5 minutes, just for this character!";
	next;
	.@buff_menu$ = "Package 1 - AGI, Blessing, Angelus (2 Nyangvine):"
		+ "Package 2 - Package 1 + Magnificat, Kyrie (3 Nyangvine):"
		+ "Package 3 - Package 2 + Impositio, Assumptio, Suffragium (5 Nyangvine):"
		+ "Package 4 - Package 3 + Wind Walk, Power-Thrust, Weapon Perfection (7 Nyangvine):"
		+ "Package 5 - Package 4 + your Soul Link (10 Nyangvine):"
		+ "Never mind:";
	switch (select(.@buff_menu$)) {
	case 1:
		callsub S_BuyBuff, 1, 2, "Package 1";
		break;
	case 2:
		callsub S_BuyBuff, 2, 3, "Package 2";
		break;
	case 3:
		callsub S_BuyBuff, 3, 5, "Package 3";
		break;
	case 4:
		callsub S_BuyBuff, 4, 7, "Package 4";
		break;
	case 5:
		callsub S_BuyBuff, 5, 10, "Package 5";
		break;
	default:
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
	}
	goto L_End;

S_ExtrasMenu:
	cutin "prm_1",2;
	mes "[Premi]";
	mes "Extra options last 24 hours for every character on this account!";
	next;
	mes "[Premi]";
	mes "Toggle the extras you want to Enable, then choose Apply selected.";
	next;
	mes "[Premi]";
	mes "Enabled extras restart at 24 hours. Disabled extras keep their current effects and time.";
	next;
	setarray .@extra_names$[0], "EXP UP", "ITEM UP", "DEATH DOWN", "TREASURE MAGNET", "IGNORE ITEM DROP PENALTY";
	setarray .@extra_costs[0], 2, 3, 1, 5, 5;
	.@extra_mask = 0;
	while (1) {
		.@menu$ = "";
		.@total = 0;
		for (.@i = 0; .@i < 5; ++.@i) {
			.@enabled = (.@extra_mask & (1 << .@i));
			.@menu$ += .@extra_names$[.@i] + " [" + (.@enabled ? "Enabled" : "Disabled") + "] (" + .@extra_costs[.@i] + " Nyangvine):";
			if (.@enabled)
				.@total += .@extra_costs[.@i];
		}
		.@menu$ += "Apply selected (" + .@total + " Nyangvine):Never mind";
		.@choice = select(.@menu$);
		if (.@choice <= 5) {
			.@extra_mask = .@extra_mask ^ (1 << (.@choice - 1));
			continue;
		}
		if (.@choice == 7)
			goto L_End;
		if (.@extra_mask == 0) {
			mes "[Premi]";
			mes "Enable at least one extra first!";
			next;
			continue;
		}
		// No dialog yield between this recheck, the single debit and delivery.
		if (callfunc("F_PremiumRemainingSeconds") <= 0) {
			mes "[Premi]";
			mes "Your Premium Buff expired. No Nyangvine was charged.";
			goto L_End;
		}
		if (Hp <= 0) {
			mes "[Premi]";
			mes "You need to be alive to buy extras. No Nyangvine was charged.";
			goto L_End;
		}
		if (countitem(6909) < .@total) {
			mes "[Premi]";
			mes "You need " + .@total + " Nyangvine for the selected extras.";
			next;
			continue;
		}
		delitem 6909, .@total;
		callfunc("F_PremiumApplyExtras", .@extra_mask);
		cutin "prm_3",2;
		mes "[Premi]";
		mes "Yay! Your selected extras are running for a fresh 24 hours!";
		goto L_End;
	}

S_BuyBuff:
	.@tier = getarg(0);
	.@cost = getarg(1);
	.@label$ = getarg(2);
	if (getstatus(SC_AUTOBUFF)) {
		mes "[Premi]";
		mes "Auto Buff is already active. Cancel it before buying another package.";
		return;
	}
	.@duration = 0;
	.@fee = 0;
	.@mode = select("One-time buff:Enable Auto Buff:Never mind");
	if (.@mode == 3) return;
	if (.@mode == 2) {
		switch (select("30 minutes (20 Nyangvine opening fee):1 hour (40 Nyangvine opening fee):3 hours (80 Nyangvine opening fee):Never mind")) {
		case 1: .@duration = 1800; .@fee = 20; break;
		case 2: .@duration = 3600; .@fee = 40; break;
		case 3: .@duration = 10800; .@fee = 80; break;
		default: return;
		}
	}
	cutin "prm_2",2;
	mes "[Premi]";
	mes .@label$ + " costs " + .@cost + " Nyangvine for 5 minutes.";
	next;
	if (.@duration) {
		mes "[Premi]";
		mes "Auto Buff lasts " + (.@duration / 60) + " minutes. Opening fee: " + .@fee + " Nyangvine.";
		next;
		mes "[Premi]";
		mes "Pay " + (.@fee + .@cost) + " Nyangvine now, including the first package.";
		next;
		mes "[Premi]";
		mes "Each renewal costs " + .@cost + " Nyangvine, 5-10 seconds after its 5-minute round ends.";
		next;
	}
	if (select("Pay:Never mind") == 2) {
		cutin "prm_5",2;
		mes "[Premi]";
		mes "Ah... o-okay. Never mind then.";
		return;
	}
	if (getstatus(SC_AUTOBUFF)) {
		mes "[Premi]";
		mes "Auto Buff is already active. Cancel it before buying another package.";
		return;
	}
	.@spirit = premibuffbuy(.@tier, .@duration);
	if (.@spirit < 0) {
		cutin "prm_4",2;
		mes "[Premi]";
		if (.@spirit == -4)
			mes "Auto Buff is already active. Cancel it before buying another package.";
		else if (.@spirit == -5)
			mes "You don't have enough Nyangvine for this purchase.";
		else if (.@spirit == -3)
			mes "Auto Buff stopped: Premium expired.";
		else if (.@spirit == -2)
			mes "You must be alive to buy a buff package.";
		else
			mes "The purchase could not be completed. Please try again.";
		return;
	}
	cutin "prm_3",2;
	mes "[Premi]";
	mes "Yay! " + .@label$ + " is running for 5 minutes!";
	if (.@duration) {
		next;
		mes "[Premi]";
		mes "Auto Buff is active. Cancel it here whenever you want.";
	}
	if (.@tier == 5 && !.@spirit) {
		next;
		cutin "prm_4",2;
		mes "[Premi]";
		mes "Your job doesn't have a classic Soul Link...";
		next;
		mes "[Premi]";
		mes "You paid for Package 5 without checking first. That's on you!";
	}
	return;

L_End:
	close2;
	cutin "",255;
	end;
