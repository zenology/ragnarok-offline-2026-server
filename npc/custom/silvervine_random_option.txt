function	script	F_SVE_StripeTierCosts	{
	.@slot = getarg(0);

		.@slot = 1;

	set getarg(1), (.@slot == 1 ? 5 : .@slot == 2 ? 7 : .@slot == 3 ? 10 : 5);

// Shared generic slices, plus Crimson's categorical O1. Outputs are references.
function	script	F_SVE_RollSharedRo	{
	set getarg(3), 0;
	set getarg(4), 0;
	set getarg(5), 0;
	.@item = getarg(0);
	.@index = getarg(1);
	.@force_max = getarg(2);
	if (!getnpcid(0, "shared_random_option_db") || .@index < 0 || .@index > 4) return 0;
	.@type = getiteminfo(.@item, ITEMINFO_TYPE);
	if (.@type != IT_WEAPON && .@type != IT_ARMOR) return 0;
	if (.@index == 0 && inarray(getvariableofnpc(.CrimsonItemId[0], "shared_random_option_db"), .@item) != -1) {
		.@roll = rand(100);
		for (.@i = 0; .@i < 10; ++.@i) {
			.@roll -= getelementofarray(getvariableofnpc(.CrimsonOptionWeight[0], "shared_random_option_db"), .@i);
			if (.@roll < 0) {
				set getarg(3), getelementofarray(getvariableofnpc(.CrimsonOptionId[0], "shared_random_option_db"), .@i);
				return 1;
			}
		}
		return 0;
	}
	.@pool = (.@type == IT_WEAPON ? 0 : 5) + .@index;
	.@start = getelementofarray(getvariableofnpc(.PoolStart[0], "shared_random_option_db"), .@pool);
	.@count = getelementofarray(getvariableofnpc(.PoolCount[0], "shared_random_option_db"), .@pool);
	if (.@count < 1 || .@start < 0 || .@start + .@count > 368) return 0;
	.@entry = .@start + rand(.@count);
	.@id = getelementofarray(getvariableofnpc(.OptionId[0], "shared_random_option_db"), .@entry);
	.@min = getelementofarray(getvariableofnpc(.OptionMin[0], "shared_random_option_db"), .@entry);
	.@max = getelementofarray(getvariableofnpc(.OptionMax[0], "shared_random_option_db"), .@entry);
	if (.@id < 1 || .@id > 32767 || .@min < -32768 || .@max > 32767 || .@min > .@max) return 0;
	set getarg(3), .@id;
	if (.@force_max)
		set getarg(4), .@max;
	else
		set getarg(4), .@min + rand(.@max - .@min + 1);
	return 1;
}
