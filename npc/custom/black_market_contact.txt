//===== Ragnarok Offline =====================================
//= Black Market Contacts
//===== Description: =========================================
//= Persistent character-local entry records for the Black Market.
//= Permit and quest gating remain deferred.
//============================================================

function	script	F_BlackMarketContact	{
	mes "[Black Market Contact]";
	mes "Keep your voice down. I work for the Black Market.";
	mes "Normally, you need a Permit before anyone lets you through.";
	mes "I can get you inside for free this time.";
	next;
	if (select("Enter the Black Market.:Not now.") == 2)
		close;

	// Permanent character variable: overwritten on every confirmed entry.
	BM_ReturnLocation = getarg(0);
	mes "[Black Market Contact]";
	mes "Stay close and do not ask questions.";
	close2;
	warp "paramk",11,171;
	end;
}

	mes "[Black Market Contact]";
	mes "Leaving already? I can put you back on the street without anyone noticing.";
	next;
	if (select("Leave quietly.:Stay here.") == 2)
		close;
